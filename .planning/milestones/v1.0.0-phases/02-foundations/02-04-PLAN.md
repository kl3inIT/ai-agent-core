**Create `020-ai-message.xml`** (mirror 040-order.xml FK pattern):

```xml
<?xml version="1.0" encoding="UTF-8"?>
<databaseChangeLog
        xmlns="http://www.liquibase.org/xml/ns/dbchangelog"
        xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
        xsi:schemaLocation="http://www.liquibase.org/xml/ns/dbchangelog
                      http://www.liquibase.org/xml/ns/dbchangelog/dbchangelog-latest.xsd">

    <changeSet id="1" author="ai-agent">
        <createTable tableName="AI_AGENT_MESSAGE">
            <column name="ID" type="${uuid.type}">
                <constraints primaryKey="true" nullable="false"/>
            </column>
            <column name="VERSION" type="int" defaultValueNumeric="1">
                <constraints nullable="false"/>
            </column>
            <column name="CONVERSATION_ID" type="${uuid.type}">
                <constraints nullable="false"
                             foreignKeyName="FK_AI_AGENT_MESSAGE__ON_CONVERSATION"
                             references="AI_AGENT_CONVERSATION(ID)"/>
            </column>
            <column name="ROLE_" type="varchar(16)"/>
            <column name="CONTENT" type="clob"/>
            <column name="SEQ" type="int"/>
            <column name="CREATED_DATE" type="datetime"/>
        </createTable>
    </changeSet>

    <changeSet id="2" author="ai-agent">
        <createIndex indexName="IDX_AI_AGENT_MESSAGE__ON_CONVERSATION"
                     tableName="AI_AGENT_MESSAGE">
            <column name="CONVERSATION_ID"/>
        </createIndex>
    </changeSet>
</databaseChangeLog>
```

Note: `clob` Liquibase type normalizes to `LONGVARCHAR` on HSQLDB and `TEXT` on Postgres. `${uuid.type}` is resolved by Jmix/EclipseLink (UUID on Postgres, varchar(36) on HSQLDB). `datetime` aligns with Jmix audit columns (OffsetDateTime → TIMESTAMP-compatible type across DBs).
