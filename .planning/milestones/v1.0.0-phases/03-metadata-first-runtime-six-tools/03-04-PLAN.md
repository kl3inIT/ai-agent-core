package com.vn.agent.tools;

import org.junit.jupiter.api.Test;
import org.objectweb.asm.*;
import org.springframework.ai.tool.annotation.Tool;

import java.io.InputStream;
import java.util.*;
import static org.assertj.core.api.Assertions.assertThat;

class BuiltInDataToolsReadOnlyTest {

    private static final Set<String> FORBIDDEN_DATA_MANAGER_METHODS = Set.of("save", "saveContext", "remove");
    private static final String DATA_MANAGER_OWNER = "io/jmix/core/DataManager";
    private static final String ENTITY_MANAGER_OWNER = "jakarta/persistence/EntityManager";

    @Test
    void noMutationPathsInToolBodies() throws Exception {
        byte[] bytes = readBytes(BuiltInDataTools.class);
        List<String> violations = new ArrayList<>();
        Set<String> toolMethods = discoverToolAnnotatedMethods();

        ClassReader cr = new ClassReader(bytes);
        cr.accept(new ClassVisitor(Opcodes.ASM9) {
            @Override public MethodVisitor visitMethod(int access, String name, String descriptor,
                                                      String signature, String[] exceptions) {
                if (!toolMethods.contains(name)) return null;
                return new MethodVisitor(Opcodes.ASM9) {
                    @Override public void visitMethodInsn(int opcode, String owner, String n, String desc, boolean itf) {
                        if (DATA_MANAGER_OWNER.equals(owner) && FORBIDDEN_DATA_MANAGER_METHODS.contains(n)) {
                            violations.add("BuiltInDataTools." + name + " calls " + owner + "." + n);
                        }
                        if (ENTITY_MANAGER_OWNER.equals(owner)) {
                            violations.add("BuiltInDataTools." + name + " calls EntityManager." + n);
                        }
                    }
                };
            }
        }, 0);

        assertThat(violations).as("TOOL-08 read-only enforcement — forbidden instructions found").isEmpty();
    }

    @Test
    void noJpqlStringBuiltFromMethodParameters() throws Exception {
        // Walk each @Tool method; track whether an ALOAD/ILOAD of a parameter local slot
        // feeds a makeConcatWithConstants or StringBuilder.append call that then flows into
        // LoadContext.createQuery(String) / DataManager.loadValue(String,...) /
        // DataManager.load(String) / EntityManager.createQuery(String).
        // Implementation sketch:
        //   - For each method, read method descriptor; arg slots are 1..N (0 is `this`).
        //   - Visit instructions; when we see INVOKEDYNAMIC with bsm "makeConcatWithConstants",
        //     look backward in the instruction stream (or collect stack-source data via the
        //     preceding ALOAD/ILOAD instructions — ASM's Analyzer + SourceInterpreter makes this robust).
        //   - If any stack element at the time of INVOKEDYNAMIC came from an ALOAD of a parameter slot,
        //     and the concat result flows into a createQuery-style INVOKEVIRTUAL/INVOKEINTERFACE, FAIL.
        //
        // Simpler alternative used for v1: scan the .class for any LDC "select" / "SELECT" / "delete" / "update" / "DELETE" / "UPDATE"
        // string constant AND an ALOAD_1/2/3 (parameter) followed by makeConcatWithConstants within the same
        // method. If both conditions hold, FAIL.
        //
        // The count_records method uses `"select e from " + mc.getName() + " e"` — `mc` is a LOCAL VARIABLE
        // assigned from a method call (resolveOrError), not directly a parameter. The test must allow a
        // makeConcatWithConstants whose non-constant operands are all local-variable slots that were assigned
        // from method calls (never directly from an incoming parameter).
    }

    private static byte[] readBytes(Class<?> c) throws Exception {
        try (InputStream in = c.getClassLoader().getResourceAsStream(c.getName().replace('.', '/') + ".class")) {
            return in.readAllBytes();
        }
    }

    private static Set<String> discoverToolAnnotatedMethods() {
        Set<String> names = new LinkedHashSet<>();
        for (var m : BuiltInDataTools.class.getDeclaredMethods()) {
            if (m.isAnnotationPresent(Tool.class)) names.add(m.getName());
        }
        return names;
    }
}
