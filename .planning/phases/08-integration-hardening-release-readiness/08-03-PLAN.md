| First-run actual | Action |
|------------------|--------|
| Equals baseline | Lock the assertion; this is the contract going forward. |
| Differs by ≤1 | Update the constant in the test (e.g. `==2` → `==3`); document the rationale in @DisplayName. NOT a bug. |

    ```java
    import javax.sql.DataSource;

    import net.ttddyy.dsproxy.support.ProxyDataSourceBuilder;
    import org.springframework.beans.BeansException;
    import org.springframework.beans.factory.config.BeanPostProcessor;
    import org.springframework.boot.test.context.TestConfiguration;
    import org.springframework.context.annotation.Bean;

    /**
     * R-03a (cross-AI review fix): wraps the EXISTING primary DataSource in-place via
     * BeanPostProcessor — does NOT register a new @Primary bean. The previous shape used
     * `@Bean @Primary DataSource queryCountingDataSource(...)` which would displace
     * whatever was previously primary in the test profile, potentially breaking the
     * agentstore datasource wiring (Jmix has TWO datasources). Wrapping in-place avoids
     * the displacement.
     *
     * <p>Pattern: BeanPostProcessor#postProcessAfterInitialization intercepts the bean
     * named "dataSource" (the Jmix main store). The agentstoreDataSource bean is
     * untouched.
     *
     * <p>Phase 8 plan 08-03 — EclipseLink-friendly substitute for SessionFactory.getStatistics
     * named in CONTEXT D-07.
     */
    @TestConfiguration
    public class QueryCountingDataSourceConfiguration {

        public static final String DATASOURCE_NAME = "counting-ds";

        @Bean
        public BeanPostProcessor queryCountingDataSourceWrapper() {
            return new BeanPostProcessor() {
                @Override
                public Object postProcessAfterInitialization(Object bean, String beanName) throws BeansException {
                    // Match the main-store DataSource by bean name. Confirm the actual
                    // primary name from AITestConfiguration before locking; "dataSource" is
                    // the Jmix convention for the main store. Do NOT wrap "agentstoreDataSource".
                    if ("dataSource".equals(beanName) && bean instanceof DataSource delegate) {
                        return ProxyDataSourceBuilder.create(delegate)
                                .name(DATASOURCE_NAME)
                                .countQuery(new SingleQueryCountHolder())
                                .build();
                    }
                    return bean;
                }
            };
        }
    }
    ```
