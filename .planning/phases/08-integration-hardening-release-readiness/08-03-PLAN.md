    ```java
    package com.vn.agent.performance;

    import javax.sql.DataSource;
    import net.ttddyy.dsproxy.listener.SingleQueryCountHolder;
    import net.ttddyy.dsproxy.support.ProxyDataSourceBuilder;
    import org.springframework.boot.test.context.TestConfiguration;
    import org.springframework.context.annotation.Bean;
    import org.springframework.context.annotation.Primary;

    /**
     * Wraps the production DataSource with a query-counting proxy. @Import this from
     * any perf test that needs per-method JDBC SELECT counts. The wrapping is
     * transparent — Jmix/EclipseLink see a normal DataSource.
     *
     * Phase 8 plan 08-03 — EclipseLink-friendly substitute for SessionFactory.getStatistics
     * named in CONTEXT D-07.
     */
    @TestConfiguration
    public class QueryCountingDataSourceConfiguration {

        @Bean
        @Primary
        public DataSource queryCountingDataSource(
                @org.springframework.beans.factory.annotation.Qualifier("dataSource") DataSource delegate) {
            return ProxyDataSourceBuilder.create(delegate)
                    .name("counting-ds")
                    .countQuery(new SingleQueryCountHolder())
                    .build();
        }
    }
    ```

    **Note on bean qualifier:** Jmix uses two stores in production (main + agentstore). The qualifier `"dataSource"` targets the main store. If the test profile uses a different bean name (read `AITestConfiguration` first), match that name. The agentstore datasource is NOT wrapped here — perf assertions target tool execution, which goes through DataManager → main store for the demo Customer/Order entities.

    If the executor finds the test profile already declares a `@Primary DataSource` bean, this @Configuration may collide. In that case: (a) make the new bean `@ConditionalOnMissingBean(name = "queryCountingDataSource")`, OR (b) wire the wrapping inside the existing @Configuration via a `@Bean BeanPostProcessor` that wraps on `postProcessAfterInitialization`. Pick (a) — simpler.
