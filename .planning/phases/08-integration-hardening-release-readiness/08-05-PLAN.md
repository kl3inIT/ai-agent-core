    import static org.assertj.core.api.Assertions.assertThat;

    import org.junit.jupiter.api.DisplayName;

    import org.springframework.beans.factory.annotation.Autowired;

    /**
     * Phase 8 plan 08-05 — TEST-07 clean-consumer smoke.
     *
     * <p>Asserts that adding `implementation 'com.vn:ai-agent-starter:<version>'`
     * to a fresh Jmix project is sufficient for ChatService to be wired and Jmix
     * views to register via the add-on's @JmixModule classpath scanning.
     *
     * <p>R-05c: AIAutoConfiguration FQN verified — com.vn.autoconfigure.agent.AIAutoConfiguration
     * (NOT com.vn.agent.autoconfigure.AIAutoConfiguration which would silently
     * fall through Class.forName and mask boot failures as test failures).
     */
    @SpringBootTest(
            classes = ConsumerSmokeApplication.class,
            webEnvironment = SpringBootTest.WebEnvironment.NONE)
    class BootSmokeTest {

        // R-05c: VERIFIED FQN. Read the actual package line of AIAutoConfiguration.java
        // before locking this literal. If the package differs, substitute the actual value.
        private static final String AI_AUTO_CONFIGURATION_FQN =
                "com.vn.autoconfigure.agent.AIAutoConfiguration";

        // I-02: hard contract — ai-agent's ChatView. Substring matching could pass on
        // a host's own ChatRoom view; FQN cannot.
        private static final String CHAT_VIEW_FQN = "com.vn.agent.view.chat.ChatView";

        @Autowired(required = false) ChatService chatService;
        @Autowired ApplicationContext applicationContext;

        @Test
        @DisplayName("ChatService bean is auto-wired by ai-agent-starter")
        void chatServiceBeanIsWired() {
            assertThat(chatService)
                    .as("ChatService must be wired by ai-agent-starter's auto-configuration "
                        + "(META-INF/spring/org.springframework.boot.autoconfigure.AutoConfiguration.imports). "
                        + "Null = the starter is not on classpath OR auto-config did not fire.")
                    .isNotNull();
        }

        @Test
        @DisplayName("ai-agent views are present as Spring beans (proves @JmixModule scan fired)")
        void aiAgentViewsRegisterAsBeans() {
            String[] viewControllerBeans = applicationContext
                    .getBeanNamesForAnnotation(io.jmix.flowui.view.ViewController.class);
            assertThat(viewControllerBeans)
                    .as("Jmix view-controller beans should include ai-agent views (ChatView et al.). "
                        + "Found beans: %s", java.util.Arrays.toString(viewControllerBeans))
                    .anyMatch(name -> name.toLowerCase().contains("chat"))
                    .anyMatch(name -> name.toLowerCase().contains("conversation"));
        }

        @Test
        @DisplayName("R-05c: AIAutoConfiguration class loaded (verified FQN com.vn.autoconfigure.agent.AIAutoConfiguration)")
        void aiAutoConfigurationLoaded() throws Exception {
            // R-05c: FQN-pinned. If Class.forName throws, the test fails LOUD with the
            // exact class name we expected — no silent fall-through to "containsBean".
            Class<?> aiAutoConfigClass = Class.forName(AI_AUTO_CONFIGURATION_FQN);
            String[] beans = applicationContext.getBeanNamesForType(aiAutoConfigClass);
            assertThat(beans)
                    .as("AIAutoConfiguration (FQN %s) must be loaded as a Spring bean", AI_AUTO_CONFIGURATION_FQN)
                    .isNotEmpty();
        }

        @Test
        @DisplayName("ai-agent ChatView bean is registered by fully-qualified class name (I-02 hard contract)")
        void aiAgentChatViewBeanIsByFQN() throws Exception {
            Class<?> chatViewClass = Class.forName(CHAT_VIEW_FQN);
            String[] chatViewBeans = applicationContext.getBeanNamesForType(chatViewClass);
            assertThat(chatViewBeans)
                    .as("FQN %s must be registered as a Spring bean by ai-agent-starter "
                        + "auto-config / @JmixModule scanning. Empty = the add-on's view package was not scanned.",
                        CHAT_VIEW_FQN)
                    .isNotEmpty();
        }
    }
