    ```java
    package com.vn.consumersmoke;

    import static org.assertj.core.api.Assertions.assertThat;

    import org.junit.jupiter.api.DisplayName;
    import org.junit.jupiter.api.Test;
    import org.springframework.beans.factory.annotation.Autowired;
    import org.springframework.boot.test.context.SpringBootTest;
    import org.springframework.context.ApplicationContext;

    import com.vn.agent.ChatService;

    /**
     * Phase 8 plan 08-05 — TEST-07 clean-consumer smoke.
     *
     * <p>Asserts that adding `implementation 'com.vn:ai-agent-starter:<version>'`
     * to a fresh Jmix project is sufficient for ChatService to be wired and Jmix
     * views to register via the add-on's @JmixModule classpath scanning.
     *
     * <p>Per CONTEXT § Claude's Discretion: bean-presence assertion preferred over
     * Vaadin RouteConfiguration enumeration (RESEARCH A2 — RouteConfiguration may
     * require a live VaadinService that is absent from WebEnvironment.NONE).
     */
    @SpringBootTest(
            classes = ConsumerSmokeApplication.class,
            webEnvironment = SpringBootTest.WebEnvironment.NONE)
    class BootSmokeTest {

        @Autowired(required = false) ChatService chatService;
        @Autowired ApplicationContext applicationContext;

        @Test
        @DisplayName("ChatService bean is auto-wired by ai-agent-starter")
        void chatServiceBeanIsWired() {
            assertThat(chatService)
                    .as("ChatService must be wired by ai-agent-starter's auto-configuration "
                        + "(META-INF/spring/org.springframework.boot.autoconfigure.AutoConfiguration.imports). "
                        + "Null = the starter is not on classpath OR auto-config did not fire.")
                    .isNotNull();
        }

        @Test
        @DisplayName("ai-agent views are present as Spring beans (proves @JmixModule scan fired)")
        void aiAgentViewsRegisterAsBeans() {
            // Jmix views are Spring beans annotated with @ViewController. The add-on's
            // ChatView, ConversationListView, ParametersListView, KnowledgeBaseView,
            // AiAuditEventListView all register as beans during context startup. Probe
            // by class name (no compile-time dependency on the FlowUI module).
            String[] viewControllerBeans = applicationContext
                    .getBeanNamesForAnnotation(io.jmix.flowui.view.ViewController.class);
            assertThat(viewControllerBeans)
                    .as("Jmix view-controller beans should include ai-agent views (ChatView et al.). "
                        + "Found beans: %s", java.util.Arrays.toString(viewControllerBeans))
                    .anyMatch(name -> name.toLowerCase().contains("chat"))
                    .anyMatch(name -> name.toLowerCase().contains("conversation"));
        }

        @Test
        @DisplayName("AI auto-configuration class loaded")
        void aiAutoConfigurationLoaded() {
            // Belt-and-suspenders: the auto-config bean itself proves AutoConfiguration.imports fired.
            assertThat(applicationContext.containsBean("AIAutoConfiguration")
                    || applicationContext.getBeanNamesForType(
                            Class.forName("com.vn.autoconfigure.agent.AIAutoConfiguration")).length > 0)
                    .as("AIAutoConfiguration must be loaded as a Spring bean")
                    .isTrue();
        }

        @Test
        @DisplayName("ai-agent ChatView bean is registered by fully-qualified class name (I-02 hard contract)")
        void aiAgentChatViewBeanIsByFQN() throws Exception {
            // I-02: substring matching on bean names (chat / conversation) is robust to
            // camelCase but produces false positives if the host project adds its own
            // ChatRoom view. The hard contract for a clean-consumer smoke is that the
            // ai-agent add-on's ChatView is present *by FQN* — host beans cannot satisfy
            // this assertion. The substring check above remains as a soft probe.
            Class<?> chatViewClass = Class.forName("com.vn.agent.view.chat.ChatView");
            String[] chatViewBeans = applicationContext.getBeanNamesForType(chatViewClass);
            assertThat(chatViewBeans)
                    .as("FQN com.vn.agent.view.chat.ChatView must be registered as a Spring bean by ai-agent-starter "
                        + "auto-config / @JmixModule scanning. Empty = the add-on's view package was not scanned.")
                    .isNotEmpty();
        }
    }
    ```

    **Notes:**
    1. The third assertion uses `Class.forName` so the test compiles even if the executor's IDE doesn't have the published artifact resolved at edit time.
    2. The view-name substring check (`"chat"`, `"conversation"`) tolerates Spring bean naming conventions (camelCase classname → first-letter-lower bean id) without coupling to exact bean ids.
    3. RESEARCH §Claude's Discretion: if `ViewController` class import path differs in your Jmix 2.8 build, `grep -r "@ViewController" ai-agent/ai-agent/src/main/java/com/vn/agent/view/` reveals the canonical FQN.
    4. DO NOT assert HTTP routes (no VaadinService in `WebEnvironment.NONE`).
    5. The test is invoked by the `bootRunSmoke` Gradle task per File B in Task 1.
