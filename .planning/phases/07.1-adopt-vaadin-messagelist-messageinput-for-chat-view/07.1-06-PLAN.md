    ```java
    package com.vn.agent.view.chat;

    import com.vn.agent.rag.CancellationRegistry;
    import org.junit.jupiter.api.Test;
    import org.mockito.Mockito;
    import reactor.core.Disposable;

    import java.util.UUID;

    import static org.mockito.ArgumentMatchers.any;
    import static org.mockito.ArgumentMatchers.eq;
    import static org.mockito.Mockito.never;
    import static org.mockito.Mockito.verify;

    /**
     * UI-02 — Stop semantics test (Phase 7.1 rewrite).
     * Asserts D-04 contract: Stop MUST route through CancellationRegistry.cancel(runId),
     * NEVER directly call Disposable.dispose() — because cancel(UUID) is where the
     * CANCELLED audit row is written (via ToolCallbackAuditDecorator chain).
     */
    class ChatViewStopTest {

        @Test
        void stopClick_callsRegistryCancel_withPreAllocatedRunId() {
            CancellationRegistry registry = Mockito.spy(new CancellationRegistry());
            UUID runId = UUID.randomUUID();
            Disposable disposable = Mockito.mock(Disposable.class);
            registry.register(runId, disposable);

            // Simulate Fragment's onStopClick path:
            registry.cancel(runId);

            verify(registry).cancel(runId);
            verify(disposable).dispose();   // registry internally disposes — that's fine
        }

        @Test
        void stopClick_doesNotBypassRegistry_neverCallsDisposableDirectly() {
            // Sentinel test for D-04 contract: grep the Fragment source for bare `.dispose()` — zero hits outside onDetach fallback.
            // Since we cannot instantiate the Fragment without Vaadin, this test grep-asserts the invariant.
            String fragmentSource = new String(
                java.nio.file.Files.readAllBytes(
                    java.nio.file.Paths.get(
                        "src/main/java/com/vn/agent/view/chat/fragment/ChatPanelFragment.java")),
                java.nio.charset.StandardCharsets.UTF_8);

            // Count raw `activeStream.dispose()` occurrences.
            int rawDisposeCount = 0;
            int idx = 0;
            while ((idx = fragmentSource.indexOf("activeStream.dispose()", idx)) != -1) {
                rawDisposeCount++;
                idx++;
            }
            // At most 1 allowed — the onDetach fallback when activeRunId == null.
            org.assertj.core.api.Assertions.assertThat(rawDisposeCount)
                .as("Stop MUST go through cancellationRegistry.cancel(runId); raw dispose permitted only in onDetach null-runId fallback")
                .isLessThanOrEqualTo(1);

            // AND cancellationRegistry.cancel() must appear at least twice (Stop click path + onDetach path).
            int cancelCount = 0;
            idx = 0;
            while ((idx = fragmentSource.indexOf("cancellationRegistry.cancel(", idx)) != -1) {
                cancelCount++;
                idx++;
            }
            org.assertj.core.api.Assertions.assertThat(cancelCount)
                .as("Stop path and onDetach path must both route through cancellationRegistry.cancel")
                .isGreaterThanOrEqualTo(2);
        }
    }
    ```
