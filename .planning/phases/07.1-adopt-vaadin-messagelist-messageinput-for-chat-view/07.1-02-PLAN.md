    2. A public static method:
    ```java
    public static String renderStreamEvent(StreamingEvent event,
                                           Map<String,String> labels,
                                           CitationState citationState) {
        return switch (event) {
            case StreamingEvent.Content c -> c.markdownChunk() == null ? "" : c.markdownChunk();
            case StreamingEvent.ToolCall tc -> "\n\n**%s**: %s".formatted(tc.toolName(), shortArgs(tc.argsJson()));
            case StreamingEvent.ToolResult tr -> {
                String outcomeLabel = labels.getOrDefault(
                    "chatView.stream.outcome." + tr.outcome().name(), tr.outcome().name());
                yield "  \n_%s — %s_\n\n---\n".formatted(outcomeLabel, tr.summary() == null ? "" : tr.summary());
            }
            case StreamingEvent.Citation c -> {
                String prefix = citationState.consumeFirst()
                    ? "\n\n---\n**%s**".formatted(labels.getOrDefault("chatView.stream.sources", "Sources"))
                    : "";
                if (c.documentId() == null) {
                    yield prefix + "\n- source"; // A5 null-guard: unlinked bullet
                }
                yield prefix + "\n- [%s](/ai-agent/knowledge?documentId=%s)"
                    .formatted(c.documentId().toString(), c.documentId());
            }
            case StreamingEvent.Error err -> "\n\n---\n**%s:** %s".formatted(
                labels.getOrDefault("chatView.stream.error", "error"), err.messageKey());
            case StreamingEvent.Final f -> ""; // v1 — skip closing summary per RESEARCH Open Question 2
        };
    }
    ```

    static String shortArgs(String argsJson) {

    }
