       public void setConversationId(UUID cid) {
           this.conversationId = cid;
           items.clear();
           if (cid == null) {
               messageList.setItems(items);
               return;
           }
           String userId = currentAuthentication.getUser().getUsername();

           // ConversationNotFoundException — caught by ChatView.beforeEnter (Plan 04)
           List<AiMessage> history = dataManager.load(AiMessage.class)
               .query("select m from ai_AiMessage m where m.conversation.id = :cid order by m.createdDate asc, m.seq asc")
               .parameter("cid", cid)
               .list();
           String userName = messageBundle.getMessage("chatView.message.userName");
           String aiName = messageBundle.getMessage("chatView.message.assistantName");
           for (AiMessage m : history) {
