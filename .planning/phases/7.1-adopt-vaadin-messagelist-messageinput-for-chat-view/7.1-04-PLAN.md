    **Class shape:**
    ```java
    @Route(value = "ai-agent/chat", layout = DefaultMainViewParent.class)
    @ViewController("AiAgent_Chat")
    @ViewDescriptor("chat-view.xml")
    public class ChatView extends StandardView implements BeforeEnterObserver {

        @ViewComponent private ChatPanelFragment chatPanelFragment;
        @Autowired private Dialogs dialogs;
        @Autowired private MessageBundle messageBundle;
        @Autowired private Notifications notifications;

        @Override
        public void beforeEnter(BeforeEnterEvent event) {
            QueryParameters qp = event.getLocation().getQueryParameters();
            List<String> cidParam = qp.getParameters().get("conversationId");
            if (cidParam == null || cidParam.isEmpty()) {
                chatPanelFragment.setConversationId(null);
                return;
            }
            try {
                UUID cid = UUID.fromString(cidParam.get(0));
                chatPanelFragment.setConversationId(cid);
            } catch (IllegalArgumentException | ConversationNotFoundException ex) {
                notifications.create(messageBundle.getMessage("chatView.error.generic"))
                    .withType(Notifications.Type.ERROR).show();
                chatPanelFragment.setConversationId(null);
            }
        }

        @Subscribe("newChatButton")
        public void onNewChatClick(ClickEvent<JmixButton> ev) {
            if (!chatPanelFragment.hasMessages() && !chatPanelFragment.isStreaming()) {
                chatPanelFragment.startNewChat();
                navigateCleanUrl();
                return;
            }
            dialogs.createOptionDialog()
                .withHeader(messageBundle.getMessage("chatView.newChat.confirmHeader"))
                .withText(messageBundle.getMessage("chatView.newChat.confirmText"))
                .withActions(
                    new DialogAction(DialogAction.Type.YES).withHandler(e -> {
                        chatPanelFragment.startNewChat();
                        navigateCleanUrl();
                    }),
                    new DialogAction(DialogAction.Type.NO))
                .open();
        }

        private void navigateCleanUrl() {
            UI.getCurrent().getPage().getHistory()
                .replaceState(null, RouteConfiguration.forSessionScope()
                    .getUrl(ChatView.class));
            // Alternative: UI.getCurrent().navigate(ChatView.class) — pick whichever matches
            // the Phase 7-03 established pattern in STATE.md.
        }
    }
    ```
