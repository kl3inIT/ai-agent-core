    @Subscribe("upload")
    public void onUploadSucceeded(final com.vaadin.flow.component.upload.SucceededEvent event) {
        com.vaadin.flow.component.upload.Upload uploadComponent = event.getUpload();
        // The buffer was instantiated by Jmix because of receiverType="MULTI_FILE_MEMORY_BUFFER".
        if (!(uploadComponent.getReceiver() instanceof com.vaadin.flow.component.upload.receivers.MultiFileMemoryBuffer buffer)) {
            log.warn("Upload receiver is not MultiFileMemoryBuffer; skipping persistence.");
            return;
        }
        String fileName = event.getFileName();
        try (java.io.InputStream inputStream = buffer.getInputStream(fileName)) {
            handleSucceededFile(fileName, event.getMIMEType(), event.getContentLength(), inputStream);
        } catch (java.io.IOException e) {
            log.warn("Failed to read uploaded file {}", fileName, e);
            notifications.create(messages.getMessage("chatView.attachments.upload.failed"))
                    .withType(io.jmix.flowui.Notifications.Type.ERROR)
                    .show();
        }
    }

    @Subscribe("upload")
    public void onUploadFailed(final com.vaadin.flow.component.upload.FailedEvent event) {
        notifications.create(messages.getMessage("chatView.attachments.upload.failed")
                + ": " + event.getReason().getMessage())
                .withType(io.jmix.flowui.Notifications.Type.ERROR)
                .show();
    }

    @Subscribe("upload")
    public void onUploadFileRejected(final com.vaadin.flow.component.upload.FileRejectedEvent event) {

        String key = "chatView.attachments.upload.tooLarge";

                .show();
    }
