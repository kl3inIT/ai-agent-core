    package com.vn.agent.taskfile;

    import org.junit.jupiter.api.Test;
    import java.io.IOException;
    import java.nio.file.Files;
    import java.nio.file.Path;
    import java.util.List;
    import java.util.stream.Stream;
    import static org.assertj.core.api.Assertions.assertThat;

    /**
     * TEST-16 (static): asserts that no source file under {@code com.vn.agent.taskfile.**}
     * AND no source file containing the {@code bulk_save_records} body in
     * {@code com.vn.agent.tools.mutation.BuiltInMutationTools} references any RAG / VectorStore /
     * Ingester / DocumentReader API. Phase 13's task-file pathway is structurally disjoint from
     * KB ingestion (see package-info.java).
     */
    class TaskFileNoVectorStoreSourceScannerTest {

        private static final List<String> FORBIDDEN_TOKENS = List.of(
                "IngesterManager",
                "VectorStore",
                "RetrievalAugmentationAdvisor",
                "TokenTextSplitter",
                "DocumentReader",
                "com.vn.agent.rag."
        );

        @Test
        void noForbiddenTokenInTaskfilePackage() throws IOException {
            Path root = Path.of("src/main/java/com/vn/agent/taskfile");
            assertThat(Files.exists(root)).as("taskfile package must exist").isTrue();
            try (Stream<Path> paths = Files.walk(root)) {
                paths.filter(p -> p.toString().endsWith(".java"))
                     .forEach(p -> {
                         try {
                             String body = Files.readString(p);
                             // Allow forbidden tokens to appear ONLY inside the JavaDoc DO-NOT-REFERENCE
                             // section of package-info.java — strip that block first.
                             String scanBody = stripDoNotReferenceJavaDoc(body);
                             for (String token : FORBIDDEN_TOKENS) {
                                 assertThat(scanBody)
                                         .as("File %s must not reference %s (TEST-16 invariant)", p, token)
                                         .doesNotContain(token);
                             }
                         } catch (IOException e) {
                             throw new RuntimeException(e);
                         }
                     });
            }
        }

        private static String stripDoNotReferenceJavaDoc(String body) {
            // Strip multi-line block(s) starting with "* <b>STRUCTURAL INVARIANT" or "DO NOT REFERENCE"
            // up to the closing "*/" so the package-info JavaDoc allowlists itself.
            int idx = body.indexOf("DO NOT REFERENCE");
            if (idx < 0) return body;
            int closeIdx = body.indexOf("*/", idx);
            if (closeIdx < 0) return body;
            return body.substring(0, idx) + body.substring(closeIdx);
        }
    }
