```java
package com.vn.agent.tools.link;

import com.vaadin.flow.router.Route;
import com.vn.agent.tools.ToolEntityResolver;
import com.vn.agent.tools.ToolResultFormatter;
import com.vn.agent.tools.ToolUserError;
import com.vn.agent.tools.UnknownEntityHints;
import io.jmix.core.metamodel.model.MetaClass;
import io.jmix.flowui.view.ViewInfo;
import io.jmix.flowui.view.ViewRegistry;
import org.apache.commons.lang3.StringUtils;
import org.springframework.ai.tool.annotation.Tool;
import org.springframework.ai.tool.annotation.ToolParam;
import org.springframework.boot.autoconfigure.web.ServerProperties;
import org.springframework.core.annotation.AnnotationUtils;
import org.springframework.lang.Nullable;
import org.springframework.stereotype.Component;

import java.net.URLEncoder;
import java.nio.charset.StandardCharsets;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Optional;
import java.util.UUID;

/**
 * Always-on link generator tools (independent of {@code ai-agent.tools.mutation.enabled}).
 * Produces deep-link URLs for the host shell using {@link ViewRegistry} routes
 * and {@link ServerProperties}-derived context path. Both tools enforce
 * {@code unknown_entity} opacity through {@link ToolEntityResolver} (Phase 10 R4).
 *
 * <p>Mutation result schema stays clean ({@code {outcome, entityId, instanceName}});
 * the LLM calls these tools separately when it wants to surface a verify-link.
 */
@Component
public class BuiltInLinkTools {

    private final ServerProperties serverProperties;
    private final ViewRegistry viewRegistry;
    private final ToolEntityResolver toolEntityResolver;
    private final ToolResultFormatter toolResultFormatter;

    public BuiltInLinkTools(ServerProperties serverProperties,
                            ViewRegistry viewRegistry,
                            ToolEntityResolver toolEntityResolver,
                            ToolResultFormatter toolResultFormatter) {
        this.serverProperties = serverProperties;
        this.viewRegistry = viewRegistry;
        this.toolEntityResolver = toolEntityResolver;
        this.toolResultFormatter = toolResultFormatter;
    }

    @Tool(name = "generate_entity_list_link", description = """
            MANDATORY WORKFLOW:
            1. Call list_entities first to confirm the entity name is visible.
            2. Pass the EXACT entity name (e.g. 'sample_Customer'), never a label.

            INPUT CONTRACT:
            - entityName: exact internal name from agent.entities or list_entities.

            PARAMETER FORMATS:
            - entityName: String, Jmix metaClass name (host-prefixed).

            ERROR HANDLING:
            - unknown_entity: entity is hidden, unknown, or has no list view registered.
              Recovery: call list_entities, retry with corrected name; if still
              unknown_entity, tell the user the entity is not browsable.

            STRICTNESS + EXAMPLES:
            ✓ CORRECT: generate_entity_list_link("sample_Customer") -> {"url":"/sample-app/customers"}
            ✗ INCORRECT: generate_entity_list_link("Customer")        (display label)
            ✗ INCORRECT: generate_entity_list_link("customers")       (route, not entity)
            """)
    public String generateEntityListLink(
            @ToolParam(description = "Exact entity name from list_entities; never a label or route")
            String entityName) {
        try {
            MetaClass metaClass = toolEntityResolver.resolveReadableEntityOrThrow(entityName);
            String listViewId = viewRegistry.getListViewId(metaClass);
            if (listViewId == null) {
                throw new ToolUserError("unknown_entity",
                        "no list view registered for " + entityName,
                        UnknownEntityHints.AS_LIST);
            }
            String route = findRouteForViewId(listViewId);
            if (route == null) {
                throw new ToolUserError("unknown_entity",
                        "no route resolvable for " + entityName + " list view",
                        UnknownEntityHints.AS_LIST);
            }
            Map<String, Object> payload = new LinkedHashMap<>();
            payload.put("url", contextPath() + "/" + route);
            return toolResultFormatter.toJson(payload);
        } catch (ToolUserError err) {
            return toolResultFormatter.error(err);
        }
    }

    @Tool(name = "generate_entity_detail_link", description = """
            MANDATORY WORKFLOW:
            1. Call list_entities to confirm the entity name is visible.
            2. Use the entityId returned by create_record / update_record / get_record.
            3. This tool does NOT verify the id exists — call get_record for that.

            INPUT CONTRACT:
            - entityName: exact internal name from agent.entities or list_entities.
            - entityId: UUID string from a prior tool call.

            PARAMETER FORMATS:
            - entityName: String, Jmix metaClass name.
            - entityId: hyphenated 36-char UUID string.

            ERROR HANDLING:
            - unknown_entity: entity hidden, unknown, or has no detail view route.
              Recovery: surface to user; do not retry without a valid entity name.

            STRICTNESS + EXAMPLES:
            ✓ CORRECT: generate_entity_detail_link("sample_Customer", "f1a2b3c4-...")
            ✗ INCORRECT: generate_entity_detail_link("sample_Customer", "Alice")  (id, not name)
            """)
    public String generateEntityDetailLink(
            @ToolParam(description = "Exact entity name from list_entities") String entityName,
            @ToolParam(description = "Entity id from create_record/update_record/get_record") String entityId) {
        try {
            MetaClass metaClass = toolEntityResolver.resolveReadableEntityOrThrow(entityName);
            String detailViewId = viewRegistry.getDetailViewId(metaClass);
            if (detailViewId == null) {
                throw new ToolUserError("unknown_entity",
                        "no detail view registered for " + entityName,
                        UnknownEntityHints.AS_LIST);
            }
            String route = findRouteForViewId(detailViewId);
            if (StringUtils.isEmpty(route)) {
                throw new ToolUserError("unknown_entity",
                        "no route resolvable for " + entityName + " detail view",
                        UnknownEntityHints.AS_LIST);
            }
            // Detail routes typically end in "/:id" — strip the placeholder
            UUID uuid = UUID.fromString(entityId);
            String encodedId = URLEncoder.encode(uuid.toString(), StandardCharsets.UTF_8);

            String trimmed = StringUtils.substringBeforeLast(route, "/");
            if (StringUtils.isEmpty(trimmed)) {
                trimmed = route;
            }
            Map<String, Object> payload = new LinkedHashMap<>();
            payload.put("url", contextPath() + "/" + trimmed + "/" + encodedId);
            return toolResultFormatter.toJson(payload);
        } catch (IllegalArgumentException invalidUuid) {
            return toolResultFormatter.error(new ToolUserError("parameter_conversion_error",
                    "entityId must be a UUID",
                    java.util.List.of("use the UUID returned by get_record/create_record/update_record")));
        } catch (ToolUserError err) {
            return toolResultFormatter.error(err);
        }
    }

    @Nullable
    private String findRouteForViewId(String viewId) {
        Optional<ViewInfo> viewInfo = viewRegistry.findViewInfo(viewId);
        if (viewInfo.isEmpty()) return null;
        Route routeAnnotation = AnnotationUtils.findAnnotation(
                viewInfo.get().getControllerClass(), Route.class);
        return routeAnnotation != null ? routeAnnotation.value() : null;
    }

    private String contextPath() {
        String cp = serverProperties.getServlet().getContextPath();
        return cp == null ? "" : cp;
    }
}
```
