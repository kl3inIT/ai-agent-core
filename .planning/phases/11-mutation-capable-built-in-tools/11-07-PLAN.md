package com.vn.agent.tools.mutation;

import com.fasterxml.jackson.core.JsonProcessingException;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.vn.agent.audit.AiAgentAuditProperties;
import com.vn.agent.audit.AuditFieldHasher;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Set;

/**
 * Serializes pre/post-image diffs to a JSON string for {@code resultSummary} on
 * the audit row, hashing values for attributes whose simple name is in
 * {@link AiAgentAuditProperties#resolvedSensitiveFields()} per AUD-07.
 *
 * <p>Shapes per CONTEXT.md "Claude's Discretion" + RESEARCH Q8:
 * <ul>
 *   <li>update: {@code [{"attribute","from","to"}, ...]}
 *   <li>create: {@code [{"attribute","to"}, ...]} ({@code from} omitted)
 *   <li>add/remove related: {@code {"relationship","action":"ADD"|"REMOVE","relatedId"}}
 * </ul>
 */
@Component
public class DiffSerializer {

    private final AiAgentAuditProperties auditProperties;
    private final ObjectMapper objectMapper;

    public DiffSerializer(AiAgentAuditProperties auditProperties,
                          ObjectMapper objectMapper) {
        this.auditProperties = auditProperties;
        this.objectMapper = objectMapper;
    }

    /** Update-record diff. Skips attributes where from == to. */
    public String serializeUpdateDiff(Map<String, Object> preImage,
                                      Map<String, Object> postImage) {
        Set<String> sensitive = auditProperties.resolvedSensitiveFields();
        boolean hash = auditProperties.resolvedHashSensitiveFields();
        List<Map<String, Object>> diffs = new ArrayList<>();
        for (Map.Entry<String, Object> entry : postImage.entrySet()) {
            String attr = entry.getKey();
            Object from = preImage.get(attr);
            Object to = entry.getValue();
            if (Objects.equals(from, to)) continue;
            String fromStr = stringValue(from);
            String toStr = stringValue(to);
            if (hash && sensitive.contains(attr)) {
                fromStr = AuditFieldHasher.sha256Hex(fromStr);
                toStr = AuditFieldHasher.sha256Hex(toStr);
            }
            Map<String, Object> diff = new java.util.LinkedHashMap<>();
            diff.put("attribute", attr);
            diff.put("from", fromStr);
            diff.put("to", toStr);
            diffs.add(diff);
        }
        return writeJson(diffs);
    }

    /** Create-record post-image (from omitted; entity didn't exist). */
    public String serializeCreatePostImage(Map<String, Object> postImage) {
        Set<String> sensitive = auditProperties.resolvedSensitiveFields();
        boolean hash = auditProperties.resolvedHashSensitiveFields();
        List<Map<String, Object>> entries = new ArrayList<>();
        for (Map.Entry<String, Object> entry : postImage.entrySet()) {
            String attr = entry.getKey();
            String toStr = stringValue(entry.getValue());
            if (hash && sensitive.contains(attr)) {
                toStr = AuditFieldHasher.sha256Hex(toStr);
            }
            Map<String, Object> e = new java.util.LinkedHashMap<>();
            e.put("attribute", attr);
            e.put("to", toStr);
            entries.add(e);
        }
        return writeJson(entries);
    }

    /** Related-write summary. */
    public String serializeRelated(String relationship, String action, String relatedId) {
        Map<String, Object> payload = new java.util.LinkedHashMap<>();
        payload.put("relationship", relationship);
        payload.put("action", action);
        payload.put("relatedId", relatedId);
        return writeJson(payload);
    }

    /** Serializes create/update LLM arguments for argumentsJson, hashing sensitive attribute values. */
    public String serializeEntityArgumentsJson(String entityName,
                                               String id,
                                               Map<String, Object> attributes,
                                               String idempotencyKey) {
        Map<String, Object> payload = new java.util.LinkedHashMap<>();
        payload.put("entityName", entityName);
        if (id != null) {
            payload.put("id", id);
        }
        payload.put("attributes", sanitizeAttributes(attributes));
        payload.put("idempotencyKey", idempotencyKey);
        return writeJson(payload);
    }

    /** Serializes add/remove relationship LLM arguments for argumentsJson. */
    public String serializeRelatedArgumentsJson(String entityName,
                                                String id,
                                                String relationship,
                                                String relatedId,
                                                String idempotencyKey) {
        Map<String, Object> payload = new java.util.LinkedHashMap<>();
        payload.put("entityName", entityName);
        payload.put("id", id);
        payload.put("relationship", relationship);
        payload.put("relatedId", relatedId);
        payload.put("idempotencyKey", idempotencyKey);
        return writeJson(payload);
    }

    private Map<String, Object> sanitizeAttributes(Map<String, Object> attributes) {
        Set<String> sensitive = auditProperties.resolvedSensitiveFields();
        boolean hash = auditProperties.resolvedHashSensitiveFields();
        Map<String, Object> safe = new java.util.LinkedHashMap<>();
        for (Map.Entry<String, Object> entry : attributes.entrySet()) {
            String attr = entry.getKey();
            Object val = entry.getValue();
            if (hash && sensitive.contains(attr)) {
                safe.put(attr, AuditFieldHasher.sha256Hex(stringValue(val)));
            } else {
                safe.put(attr, val);
            }
        }
        return safe;
    }

    private static String stringValue(Object v) {
        return v == null ? null : v.toString();
    }

    private String writeJson(Object value) {
        try {
            return objectMapper.writeValueAsString(value);
        } catch (JsonProcessingException e) {
            // Fail-closed: never throw from audit serialization (P-22 — audit MUST land)
            return "{}";
        }
    }
}

**STEP A — Create `MutationSaveExecutor.java` (separate @Component holding @Transactional save).**

This is mandatory: invoking a `@Transactional` method on `this` from another method in the SAME class bypasses the Spring AOP proxy ("self-invocation" pitfall) and silently runs WITHOUT a transaction. The fix is a separate `@Component` so the proxy is crossed when `BuiltInMutationTools.mutationSaveExecutor.save(...)` is called.

```java
package com.vn.agent.tools.mutation;

import io.jmix.core.DataManager;
import io.jmix.core.EntitySet;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Transactional;

/**
 * Holds the {@link Transactional @Transactional} save boundary for mutation tools.
 *
 * <p>Why a separate component instead of a private method on
 * {@link BuiltInMutationTools}? Spring's AOP transaction interceptor only
 * applies when a method call crosses the proxy. Self-invocation
 * ({@code this.doSave()} from another method on the SAME bean) bypasses the
 * proxy and runs WITHOUT any transaction — the @Transactional annotation is
 * silently ignored. Putting the save method on a different bean makes the
 * proxy boundary explicit.
 *
 * <p>Uses regular {@link DataManager} (NOT {@code UnconstrainedDataManager}) so
 * host policies and entity listeners run — MUT-03 requirement.
 */
@Component
public class MutationSaveExecutor {

    private final DataManager dataManager;

    public MutationSaveExecutor(DataManager dataManager) {
        this.dataManager = dataManager;
    }

    /** Single-entity save (create_record, update_record). */
    @Transactional
    public Object save(Object entity) {
        return dataManager.save(entity);
    }

    /** EntitySet save for parent + child writes (add_related_record, non-deleting remove_related_record). */
    @Transactional
    public EntitySet saveAll(Object... entities) {
        return dataManager.save(entities);
    }
}
```

        return AiToolCallOutcome.COMMIT_FAILED;

private void markFailedIfReserved(MutationIntentRepository.ReservationResult reservation,
                                  ToolUserError translated,
                                  MutationCommitState commitState) {
    if (reservation != null
            && reservation.state() == MutationIntentRepository.ReservationState.RESERVED) {
        try {
            if (commitState == MutationCommitState.NO_HOST_WRITE) {
                mutationIntentRepository.markFailed(reservation.intent(), translated.toDto().error());
            } else if (commitState == MutationCommitState.HOST_SAVE_RETURNED) {
                mutationIntentRepository.markCommitUnknown(reservation.intent(), translated.toDto().error());
            } else {
                // The idempotency row is already COMMITTED. Later audit/result failures must not downgrade it.
            }
        } catch (RuntimeException finalizationFailure) {
            // Preserve the original mutation error/audit path; do not leak details to the LLM.
            // If the host save returned but markCommitted failed, leaving the row PENDING is safer than marking FAILED:
            // a replay will return concurrent_modification instead of duplicating the host write.
        }
    }
}

private String replayResult(AiMutationIntent existing, String toolName,
                            long startedAt, String userUsername, String argumentsJson) {
    String freshName = null;
    try {
        if (existing.getResultEntityName() != null && existing.getResultEntityId() != null) {
            MetaClass mc = metadata.getClass(existing.getResultEntityName());
            Object loaded = dataManager.load(mc.getJavaClass())
                    .fetchPlan(FetchPlan.INSTANCE_NAME)
                    .id(existing.getResultEntityId())
                    .optional().orElse(null);
            if (loaded != null) {
                freshName = metadataTools.getInstanceName(loaded);
            }
        }
    } catch (RuntimeException ignored) {
        // entity may have been deleted; freshName stays null
    }
    Map<String, Object> result = new LinkedHashMap<>();
    result.put("outcome", AiToolCallOutcome.IDEMPOTENT_REPLAY.getId());
    result.put("entityId", existing.getResultEntityId());
    result.put("instanceName", freshName);

    safeWriteAudit(toolName, argumentsJson, null,
            System.currentTimeMillis() - startedAt,
            AiToolCallOutcome.IDEMPOTENT_REPLAY, null, null, userUsername);
    return toolResultFormatter.toJson(result);
}

private void safeWriteAudit(String eventName, String argumentsJson, String resultSummary,

        log.warn("Mutation tool audit write failed for {}", eventName, auditFailure);
    }
}

Required helper behavior:
- `requestHash(...)` builds canonical JSON from `{toolName, entityName, id, relationship, relatedId, attributes}` with sorted keys, then hashes it with a stable SHA-256 helper (`AuditFieldHasher.sha256Hex` or equivalent already present in the codebase). Never include raw PII in thrown exception messages.
- `handleReservationResult(REPLAY)` receives the current full `argumentsJson`, builds a fresh replay DTO from `resultEntityId` and `resultEntityName`, sets `outcome=IDEMPOTENT_REPLAY`, re-resolves `instanceName` live when the entity still loads, and writes the replay audit row with the current full tool arguments. Do not store or replay full result JSON; D-02 forbids it.
- `handleReservationResult(VIOLATION)` returns `MutationErrorTranslator.idempotencyViolation(...)`.
- `handleReservationResult(PENDING)` returns `concurrent_modification` with a hint to call `get_record` / `find_records` to verify state and not retry automatically while the operation is in-flight or commit outcome is unknown.
- All direct `ToolUserError` catches call `mutationErrorTranslator.translate(...)` before formatting. Only the translator decides stable-code normalization and safe-message rebuilding; mutation tools must not return a raw legacy `invalid_literal`, `unsupported_type`, or `invalid_id` code and must not pass through raw `ToolUserError` prose directly.
- All audit writes inside `BuiltInMutationTools` go through `safeWriteAudit(...)`, which catches/logs `RuntimeException` and never throws back into the tool catch ladder. This prevents recursive audit attempts and keeps `MutationToolCallbackBoundaryDecorator` from seeing in-method audit failures as callback-level failures.
- Every catch block after a RESERVED result calls `markFailedIfReserved(reservation, translated, commitState)` before returning the error. `markFailedIfReserved` reads `ToolUserError.toDto().error()`; there is no `getCode()` method. When `commitState=NO_HOST_WRITE`, it may mark FAILED and allow same-hash exact retry. When `commitState=HOST_SAVE_RETURNED`, it must mark COMMIT_UNKNOWN (or leave PENDING if even that fails) and must never mark FAILED/reclaimable. When `commitState=INTENT_COMMITTED`, it must not mutate the idempotency row at all.
- Success paths call `mutationIntentRepository.markCommitted(...)` before success audit and before returning the result, then immediately set `commitState=INTENT_COMMITTED`. If `markCommitted` throws after the host save/saveAll already returned, set audit outcome `AiToolCallOutcome.COMMIT_FAILED`, return `mutationErrorTranslator.commitFailed(...)`, and best-effort mark the reservation COMMIT_UNKNOWN; never mark it FAILED.
- `auditOutcome(Throwable, MutationCommitState)` returns `COMMIT_FAILED` only for `HOST_SAVE_RETURNED` failures, meaning the host save/saveAll returned but idempotency finalization did not reach COMMITTED. Save-time optimistic-lock, validation, and data-integrity failures happen while `commitState=NO_HOST_WRITE`, keep their stable error code, and may mark the reservation FAILED for same-hash retry. Audit-write failures after `INTENT_COMMITTED` are absorbed by `safeWriteAudit`; any other unexpected post-COMMITTED failure must leave the intent COMMITTED so exact retries replay safely.
- The `RunContext` accessor names MUST match the actual static `RunContext` API — read the file before finalizing.
- `toolEntityResolver.parseEntityId` signature is `parseEntityId(String id, MetaClass metaClass)`; all mutation call sites must use that exact argument order.
- Mutation tools are UUID-id tools in v1.1 because D-01 requires UUID id strings. `parseEntityId(...)` returns `Object`; every mutation call site must pass the result through `requireUuidId(...)` before assigning to a UUID variable or storing in `AiMutationIntent.resultEntityId`.

- `update_record` in v1.1 is not a compare-and-swap API: it does not accept `expectedVersion` and must not promise stale-read protection across a prior `get_record` turn. It only maps optimistic-lock conflicts detected during the tool's own transactional save.
