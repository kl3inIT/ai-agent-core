```java
package com.vn.agent.tools.mutation;

import com.vn.agent.tools.AgentToolCallbacks;
import org.junit.jupiter.api.Test;
import org.springframework.ai.tool.ToolCallback;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;

import java.util.Arrays;
import java.util.List;
import java.util.Set;

import static org.assertj.core.api.Assertions.assertThat;

/**
 * TEST-13 — default-config boot test. Asserts the silent default-on regression gate:
 * with no override of {@code ai-agent.tools.mutation.enabled}, the callback set
 * exposed to the chat client is exactly read tools + link tools, with ZERO mutation
 * callbacks (and especially no {@code delete_record}).
 */
@SpringBootTest
class AgentToolCallbacksDefaultConfigTest {

    @Autowired
    AgentToolCallbacks agentToolCallbacks;

    @Test
    void defaultConfig_exposesEightCallbacks_zeroMutation_zeroDeleteRecord() {
        ToolCallback[] callbacks = agentToolCallbacks.forCurrentUser();
        List<String> names = Arrays.stream(callbacks)
                .map(cb -> cb.getToolDefinition().name())
                .toList();

        // 6 read tools (BuiltInDataTools) + 2 link tools (BuiltInLinkTools) = 8
        assertThat(callbacks).hasSize(8);

        Set<String> mutationNames = Set.of(
                "create_record", "update_record",
                "add_related_record", "remove_related_record",
                "delete_record");
        assertThat(names).doesNotContainAnyElementsOf(mutationNames);

        assertThat(names).contains("generate_entity_list_link",
                "generate_entity_detail_link");
    }
}
```

```java
class BuiltInMutationToolsIdempotencyReplayTest {

    @Autowired
    BuiltInMutationTools tools;

    @Autowired
    UnconstrainedDataManager dataManager;

    @Autowired
    ObjectMapper objectMapper;

    private String idempotencyKey;
    private UUID createdEntityId;

    @Test
    @Transactional
    @Commit  // afterCommit MUST fire so MutationIntentRepository writes the dedup row
    void sameIdempotencyKey_returnsReplay_andOnlyOneDedupRow() throws Exception {
        idempotencyKey = UUID.randomUUID().toString();
        Map<String, Object> attributes = Map.of("name", "Alice");

        String secondJson = tools.createRecord("test_MutationTestFixture", attributes, idempotencyKey);

        assertThat(second.get("outcome").asText()).isEqualTo("IDEMPOTENT_REPLAY");
        // same entityId echoed back
        assertThat(second.get("entityId").asText())
                .isEqualTo(first.get("entityId").asText());

                .parameter("tool", "create_record")
                .parameter("key", idempotencyKey)
                .list();
        assertThat(dedupRows).hasSize(1);

    void cleanup() {

                    .query("select e from aiMutation_AiMutationIntent e " +
                            "where e.idempotencyKey = :key")
                    .parameter("key", idempotencyKey)
                    .list();

        if (createdEntityId != null) {
            dataManager.load(com.vn.agent.tools.mutation.fixture.MutationTestFixture.class)
                    .id(createdEntityId)
                    .optional()
                    .ifPresent(dataManager::remove);
        }
    }
```
