```java
package com.vn.agent.tools.mutation;

import io.jmix.core.Metadata;
import io.jmix.core.UnconstrainedDataManager;
import org.springframework.stereotype.Component;
import org.springframework.transaction.support.TransactionSynchronization;
import org.springframework.transaction.support.TransactionSynchronizationManager;

import java.time.Duration;
import java.time.OffsetDateTime;
import java.util.Optional;
import java.util.UUID;

/**
 * Idempotency dedup repository for {@link AiMutationIntent}.
 *
 * <p>Uses {@link UnconstrainedDataManager} per MEMORY
 * {@code feedback_jmix_unconstrained_for_system_writes}: dedup rows must be
 * readable/writable regardless of user-role grants on AiMutationIntent.
 *
 * <p>Per RESEARCH Open Question 4, {@link #recordResultAfterCommit} schedules
 * the dedup row insert via {@link TransactionSynchronization#afterCommit()} so
 * the row only persists when the host save committed. On rollback, no dedup
 * row is created — the LLM can safely retry the same idempotencyKey.
 */
@Component
public class MutationIntentRepository {

    private final UnconstrainedDataManager dataManager;
    private final Metadata metadata;

    public MutationIntentRepository(UnconstrainedDataManager dataManager,
                                    Metadata metadata) {
        this.dataManager = dataManager;
        this.metadata = metadata;
    }

    /**
     * Looks up an existing dedup row by composite key per D-02.
     * The fluent {@code load(Class)} auto-resolves the agentstore from
     * {@code @Store(name="agentstore")} on AiMutationIntent — no explicit
     * {@code .store("agentstore")} needed (per MEMORY feedback_jmix_loadvalue_store).
     */
    public Optional<AiMutationIntent> findExisting(String toolName,
                                                   String idempotencyKey,
                                                   String userUsername) {
        return dataManager.load(AiMutationIntent.class)
                .query("select e from aiMutation_AiMutationIntent e " +
                        "where e.toolName = :toolName " +
                        "and e.idempotencyKey = :key " +
                        "and e.userUsername = :user")
                .parameter("toolName", toolName)
                .parameter("key", idempotencyKey)
                .parameter("user", userUsername)
                .optional();
    }

    /**
     * Schedules a dedup row insert for after the current transaction commits.
     * Per RESEARCH Open Question 4: never write the row pre-flight (orphan
     * dedup rows on rollback would manifest as false idempotency_violation).
     */
    public void recordResultAfterCommit(String toolName,
                                        String idempotencyKey,
                                        String userUsername,
                                        UUID conversationId,
                                        UUID resultEntityId,
                                        String resultEntityName,
                                        Duration ttl) {
        if (TransactionSynchronizationManager.isSynchronizationActive()) {
            TransactionSynchronizationManager.registerSynchronization(new TransactionSynchronization() {
                @Override
                public void afterCommit() {
                    persist(toolName, idempotencyKey, userUsername,
                            conversationId, resultEntityId, resultEntityName, ttl);
                }
            });
        } else {
            // No active transaction (e.g., test path or pre-existing row replay).
            persist(toolName, idempotencyKey, userUsername,
                    conversationId, resultEntityId, resultEntityName, ttl);
        }
    }

    private void persist(String toolName,
                         String idempotencyKey,
                         String userUsername,
                         UUID conversationId,
                         UUID resultEntityId,
                         String resultEntityName,
                         Duration ttl) {
        AiMutationIntent intent = metadata.create(AiMutationIntent.class);
        intent.setToolName(toolName);
        intent.setIdempotencyKey(idempotencyKey);
        intent.setUserUsername(userUsername);
        intent.setConversationId(conversationId);
        intent.setResultEntityId(resultEntityId);
        intent.setResultEntityName(resultEntityName);
        OffsetDateTime now = OffsetDateTime.now();
        intent.setCreatedAt(now);
        intent.setExpiresAt(now.plus(ttl));
        dataManager.save(intent);
    }

    /**
     * Bulk-removes expired dedup rows. Invoked hourly by the cleanup job.
     * Returns the number of rows removed for logging/metrics.
     */
    public int deleteExpired(OffsetDateTime now) {
        java.util.List<AiMutationIntent> expired = dataManager.load(AiMutationIntent.class)
                .query("select e from aiMutation_AiMutationIntent e where e.expiresAt < :now")
                .parameter("now", now)
                .list();
        for (AiMutationIntent intent : expired) {
            dataManager.remove(intent);
        }
        return expired.size();
    }
}
```

```java
package com.vn.agent.tools.mutation;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Transactional;

import java.time.OffsetDateTime;

/**
 * Hourly TTL reaper for {@link AiMutationIntent} dedup rows. Idempotent
 * (multi-instance safe — bulk delete by expiresAt &lt; now).
 *
 * <p>Requires {@code @EnableScheduling} on AIConfiguration (added in Plan 11-02).
 * Targets the {@code agentstoreTransactionManager} since the entity lives in agentstore.
 */
@Component
public class MutationIntentCleanupJob {

    private static final Logger log = LoggerFactory.getLogger(MutationIntentCleanupJob.class);

    private final MutationIntentRepository repository;

    public MutationIntentCleanupJob(MutationIntentRepository repository) {
        this.repository = repository;
    }

    @Scheduled(cron = "0 0 * * * *")  // hourly at minute 0
    @Transactional("agentstoreTransactionManager")
    public void deleteExpiredIntents() {
        int removed = repository.deleteExpired(OffsetDateTime.now());
        if (removed > 0) {
            log.debug("Removed {} expired AiMutationIntent rows", removed);
        }
    }
}
```
