           ```java
           Flux<StreamingEvent> source = chatService.stream(userId, conversationId, text, null);

           // Split Content events from structural events so Content can be coalesced
           // without delaying ToolCall/ToolResult/Citation/Final. We use
           // Flux.groupBy + bufferTimeout on the Content group.
           activeStream = source
                   .publish(shared -> Flux.merge(
                           // Content: buffer up to 50 ms or 64 chunks, whichever comes first
                           shared.ofType(StreamingEvent.Content.class)
                                 .bufferTimeout(64, Duration.ofMillis(50))
                                 .filter(batch -> !batch.isEmpty())
                                 .map(batch -> new ContentBatch(batch)),
                           // Everything else passes through untouched
                           shared.filter(e -> !(e instanceof StreamingEvent.Content))
                   ))
                   .subscribe(
                       event -> handleEvent(event),
                       error -> handleError(error),
                       () -> finishStream()
                   );
           ```

           activeStream = source
