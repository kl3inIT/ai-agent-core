       ```java
       @Override
       public Flux<StreamingEvent> stream(String userId, UUID conversationId, String message, Overrides overrides) {
           UUID runId = UUID.randomUUID();
           long startNanos = System.nanoTime();

           return Flux.defer(() -> {
               RunContext.set(runId);
               Sinks.Many<StreamingEvent> toolSink = Sinks.many().unicast().onBackpressureBuffer();

               // Load or create conversation, enforce ownership (same path as ask())
               UUID convId = conversationGateway.loadOrCreate(userId, conversationId, message).getId();

               // Build prompt spec — reuse the same Overrides-aware prep that ask() uses,
               // extracted to a shared private helper. If the shared helper does not exist,
               // factor the existing ask() preamble (RateLimitGuard.check, TokenBudgetGuard.check,
               // AiParametersResolver.effectiveModel/SystemPrompt, ChatClient.prompt() setup)
               // into a method and call from both ask() and stream().

               Flux<StreamingEvent> content;
               try {
                   // Attempt streaming via ChatClient.prompt()...stream().chatResponse()
                   content = chatClient.prompt()
                           .messages(/* baseline + system prompt + user msg */)
                           .stream()
                           .chatResponse()
                           .<StreamingEvent>concatMap(chunk -> {
                               AssistantMessage am = chunk.getResult() != null ? chunk.getResult().getOutput() : null;
                               String text = am != null ? am.getText() : null;
                               return (text != null && !text.isEmpty())
                                       ? Flux.just(new StreamingEvent.Content(text))
                                       : Flux.empty();
                           })
                           .doOnComplete(toolSink::tryEmitComplete);
               } catch (UnsupportedOperationException nonStreaming) {
                   // D-04 graceful fallback: provider does not support streaming
                   ChatResponseDto blocking = ask(userId, convId, message, overrides);
                   toolSink.tryEmitComplete();
                   content = Flux.just(
                           new StreamingEvent.Content(blocking.content()),
                           new StreamingEvent.Final(blocking.runId(), blocking.latencyMs(), 0, 0));
               }

               Flux<StreamingEvent> merged = toolSink.asFlux().mergeWith(content);

               // Emit a Final event after content completes (only if streaming path succeeded)
               return merged.concatWith(Flux.defer(() -> {
                   long latencyMs = (System.nanoTime() - startNanos) / 1_000_000L;
                   return Flux.just(new StreamingEvent.Final(runId, latencyMs, 0, 0));
               }));
           })
           .subscribeOn(chatStreamingScheduler)
           .onErrorResume(ex -> Flux.just(mapToStreamingError(ex)))
           .doFinally(signalType -> {
               RunContext.clear();
               cancellationRegistry.clearRun(runId); // use whatever API the registry exposes; see note
           });
       }

       private StreamingEvent mapToStreamingError(Throwable ex) {
           // Map typed guard exceptions to stable i18n keys; never leak raw exception text.
           if (ex instanceof RateLimitExceededException) return new StreamingEvent.Error("chatView.error.rateLimited", Map.of());
           if (ex instanceof TokenBudgetExhaustedException) return new StreamingEvent.Error("chatView.error.tokenBudget", Map.of());
           if (ex instanceof IterationCapExceededException) return new StreamingEvent.Error("chatView.error.iterationCap", Map.of());
           if (ex instanceof ToolVetoedException) return new StreamingEvent.Error("chatView.error.toolVetoed", Map.of());
           if (ex instanceof ConversationNotFoundException) throw (ConversationNotFoundException) ex; // signal to caller
           return new StreamingEvent.Error("chatView.error.generic", Map.of());
       }
       ```
