    ```java
    package com.vn.agent.guard;

    import io.jmix.core.security.CurrentAuthentication;
    import org.springframework.cache.Cache;
    import org.springframework.cache.CacheManager;
    import org.springframework.stereotype.Component;

    import java.time.Instant;
    import java.util.ArrayDeque;
    import java.util.Deque;

    /**
     * Per-minute sliding-window rate limiter keyed by username (GUARD-01, D-13).
     * Uses a Spring Cache named "ai-agent.rateLimit" (D-12) to store a per-user
     * Deque<Long> of epoch-ms timestamps. A deque simplifies eviction of
     * entries older than 60 seconds. When the deque size >= ceiling, throws
     * RateLimitExceededException with the ceiling on a side field — the
     * super() message is the stable key "rate-limit-exceeded" (D-10 opacity).
     *
     * Disabled via jmix.ai-agent.guard.rate-limit.enabled=false — check() is a no-op.
     */
    @Component
    public class RateLimitGuard {

        private static final String CACHE_NAME = "ai-agent.rateLimit";
        private static final long WINDOW_MS = 60_000L;

        private final CacheManager cacheManager;
        private final AiAgentGuardProperties props;
        private final CurrentAuthentication currentAuthentication;

        public RateLimitGuard(CacheManager cacheManager,
                              AiAgentGuardProperties props,
                              CurrentAuthentication currentAuthentication) {
            this.cacheManager = cacheManager;
            this.props = props;
            this.currentAuthentication = currentAuthentication;
        }

        public void check() {
            if (!props.rateLimitEnabled()) return;
            String username = currentAuthentication.getUser().getUsername();
            check(username);
        }

        @SuppressWarnings("unchecked")
        public synchronized void check(String username) {
            if (!props.rateLimitEnabled()) return;
            int ceiling = props.resolvedRequestsPerMinute();
            Cache cache = cacheManager.getCache(CACHE_NAME);
            if (cache == null) return; // defensive; autoconfig ensures existence
            Deque<Long> hits = cache.get(username, Deque.class);
            long now = Instant.now().toEpochMilli();
            if (hits == null) {
                hits = new ArrayDeque<>();
            }
            // evict entries older than WINDOW_MS
            while (!hits.isEmpty() && now - hits.peekFirst() > WINDOW_MS) {
                hits.pollFirst();
            }
            if (hits.size() >= ceiling) {
                // Ceiling reached — throw. DO NOT record this attempt (avoid starvation after cooldown).
                throw new RateLimitExceededException(ceiling);
            }
            hits.offerLast(now);
            cache.put(username, hits);
        }
    }
    ```
    Note: `synchronized` on the instance means all users contend on one lock. This is acceptable for the default ConcurrentMapCacheManager; hosts swapping to Redis should replace this guard entirely (documented in SUMMARY). Alternative to consider: per-username striped lock — deferred per D-13 open question.

    ```java
    package com.vn.agent.guard;

    import org.springframework.cache.Cache;
    import org.springframework.cache.CacheManager;
    import org.springframework.stereotype.Component;

    import java.util.UUID;

    /**
     * Per-conversation token accumulator with a hard ceiling (GUARD-02, D-14).
     * Uses cache "ai-agent.tokenBreaker" (D-12) storing Long totals keyed by conversation UUID.
     * check() throws when total &gt;= ceiling; accumulate() adds post-response usage.
     * Disabled via jmix.ai-agent.guard.token-breaker.enabled=false — both methods no-op.
     */
    @Component
    public class TokenBudgetGuard {

        private static final String CACHE_NAME = "ai-agent.tokenBreaker";

        private final CacheManager cacheManager;
        private final AiAgentGuardProperties props;

        public TokenBudgetGuard(CacheManager cacheManager, AiAgentGuardProperties props) {
            this.cacheManager = cacheManager;
            this.props = props;
        }

        public synchronized void check(UUID conversationId) {
            if (!props.tokenBreakerEnabled() || conversationId == null) return;
            long total = currentTotal(conversationId);
            if (total >= props.resolvedTokenCeiling()) {
                throw new TokenBudgetExhaustedException(props.resolvedTokenCeiling());
            }
        }

        public synchronized void accumulate(UUID conversationId, long tokensUsed) {
            if (!props.tokenBreakerEnabled() || conversationId == null || tokensUsed <= 0) return;
            Cache cache = cacheManager.getCache(CACHE_NAME);
            if (cache == null) return;
            long newTotal = currentTotal(conversationId) + tokensUsed;
            cache.put(conversationId, newTotal);
        }

        /** Test hook + override for conversation reset. */
        public synchronized void reset(UUID conversationId) {
            Cache cache = cacheManager.getCache(CACHE_NAME);
            if (cache != null) cache.evict(conversationId);
        }

        private long currentTotal(UUID conversationId) {
            Cache cache = cacheManager.getCache(CACHE_NAME);
            if (cache == null) return 0L;
            Long v = cache.get(conversationId, Long.class);
            return v == null ? 0L : v;
        }
    }
    ```

    ```java
    package com.vn.agent.guard;

    /**
     * Per-turn ThreadLocal<Integer> counter for tool-execution iterations (GUARD-03).
     * Owned by GuardedToolCallingManager — set() at start of turn, increment() before each
     * tool call, reset() in the finally block. Lifecycle discipline mirrors RunContext.
     */
    public final class IterationCounter {

        private static final ThreadLocal<Integer> COUNT = new ThreadLocal<>();

        private IterationCounter() {}

        public static void start() { COUNT.set(0); }
        public static int current() {
            Integer v = COUNT.get();
            return v == null ? 0 : v;
        }
        public static int increment() {
            int next = current() + 1;
            COUNT.set(next);
            return next;
        }
        public static void reset() { COUNT.remove(); }
    }
    ```

    ```java
    import org.slf4j.Logger;
    import org.slf4j.LoggerFactory;
    import org.springframework.ai.chat.client.ChatClientRequest;
    import org.springframework.ai.chat.client.ChatClientResponse;
    import org.springframework.ai.chat.client.advisor.api.CallAdvisor;
    import org.springframework.ai.chat.client.advisor.api.CallAdvisorChain;
    import org.springframework.core.Ordered;
    import org.springframework.lang.NonNull;

    import java.util.List;

    /**
     * CallAdvisor at {@code Ordered.HIGHEST_PRECEDENCE + 400} (GUARD-05, D-17).
     * Matches assistant text against a pre-compiled set of regex patterns.
     * On match: writes the pattern KEY (never the matched text) into
     * {@code ChatClientResponse.context()} under "outputScanner.flaggedPatternKey".
     * Flag-and-pass-through — response body is unchanged; DefaultChatServiceImpl
     * promotes the context entry into ChatResponseDto.flagged + flaggedPatternKey.
     *
     * Disabled via jmix.ai-agent.guard.output-scanner.enabled=false — advisor short-circuits.
     */
    public class OutputScannerAdvisor implements CallAdvisor {

        public static final String CONTEXT_KEY_FLAGGED_PATTERN = "outputScanner.flaggedPatternKey";
        private static final Logger log = LoggerFactory.getLogger(OutputScannerAdvisor.class);

        private final AiAgentGuardProperties props;
        private final List<CompiledOutputScannerPattern> patterns;

        public OutputScannerAdvisor(AiAgentGuardProperties props) {
            this.props = props;
            this.patterns = props.resolvedPatterns().stream()
                    .map(CompiledOutputScannerPattern::from)
                    .toList();
        }

        @Override
        public @NonNull String getName() { return "OutputScannerAdvisor"; }

        @Override
        public int getOrder() { return Ordered.HIGHEST_PRECEDENCE + 400; }

        @Override
        public @NonNull ChatClientResponse adviseCall(@NonNull ChatClientRequest request,
                                                       @NonNull CallAdvisorChain chain) {
            ChatClientResponse response = chain.nextCall(request);
            if (!props.outputScannerEnabled()) return response;
            if (response == null || response.chatResponse() == null
                    || response.chatResponse().getResult() == null
                    || response.chatResponse().getResult().getOutput() == null) {
                return response;
            }
            String text = response.chatResponse().getResult().getOutput().getText();
            if (text == null || text.isEmpty()) return response;
            for (CompiledOutputScannerPattern p : patterns) {
                if (p.pattern().matcher(text).find()) {
                    try {
                        response.context().put(CONTEXT_KEY_FLAGGED_PATTERN, p.key()); // D-17: KEY only
                    } catch (Exception ex) {
                        log.debug("Failed to write scanner flag to response context", ex);
                    }
                    break; // first-match wins; audit row records exactly one pattern key per turn
                }
            }
            return response;
        }
    }
    ```
