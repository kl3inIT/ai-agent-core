        @Test
        void everyPhase6KeyExistsInBothLocales() {
            ResourceBundle en = ResourceBundle.getBundle(BUNDLE_BASENAME, Locale.ENGLISH);
            ResourceBundle vi = ResourceBundle.getBundle(BUNDLE_BASENAME, new Locale("vi"));

            for (String key : enPhase6) {

                assertFalse(viValue == null || viValue.isBlank(),
                        "VI value for key " + key + " is blank");
                assertFalse(en.getString(key).isBlank(),
                        "EN value for key " + key + " is blank");
            }
        }

        @Test
        void flaggedEnumKeyExistsInBothLocales() {
            ResourceBundle en = ResourceBundle.getBundle(BUNDLE_BASENAME, Locale.ENGLISH);
            ResourceBundle vi = ResourceBundle.getBundle(BUNDLE_BASENAME, new Locale("vi"));
            String key = "com.vn.agent.entity/AiToolCallOutcome.FLAGGED";
            assertFalse(en.getString(key).isBlank());
            assertFalse(vi.getString(key).isBlank());
        }

        private static Set<String> keysMatching(ResourceBundle bundle, String... prefixes) {
            return bundle.keySet().stream()
                    .filter(k -> {
                        for (String p : prefixes) if (k.startsWith(p)) return true;
                        return false;
                    })
                    .collect(Collectors.toCollection(TreeSet::new));
        }

        private static Set<String> symmetricDiff(Set<String> a, Set<String> b) {
            Set<String> diff = new TreeSet<>(a);
            diff.removeAll(b);
            Set<String> otherWay = new TreeSet<>(b);
            otherWay.removeAll(a);
            diff.addAll(otherWay);
            return diff;
        }
    }

        @Test void checkThrowsWhenAtOrAboveCeiling() {

            assertThrows(TokenBudgetExhaustedException.class, () -> guard.check(convId));
        }

        @Test void concurrentAccumulateIsConsistent() throws Exception {
            UUID convId = UUID.randomUUID();
            int threads = 10;
            int perThread = 100;
            ExecutorService pool = Executors.newFixedThreadPool(threads);
            CountDownLatch start = new CountDownLatch(1);
            CountDownLatch done = new CountDownLatch(threads);
            for (int i = 0; i < threads; i++) {
                pool.submit(() -> {
                    try { start.await(); } catch (InterruptedException ignored) {}
                    for (int j = 0; j < perThread; j++) guard.accumulate(convId, 1);
                    done.countDown();
                });
            }
            start.countDown();
            done.await(5, TimeUnit.SECONDS);
            pool.shutdown();

            // Force a check with an enormous expected total
            // Total should be threads * perThread = 1000 (exactly the ceiling) → check throws
            assertThrows(TokenBudgetExhaustedException.class, () -> guard.check(convId));
        }
    }
