```java
package com.vn.agent.rag;

import com.vn.agent.exposure.LlmExposurePolicy;
import com.vn.agent.rag.config.AiAgentEmbeddingProperties;
import com.vn.agent.rag.config.AiAgentRagProperties;
import com.vn.agent.security.AiAgentAdminRole;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;
import org.springframework.ai.vectorstore.filter.Filter;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.authority.SimpleGrantedAuthority;

import java.util.List;
import java.util.Set;

import static org.assertj.core.api.Assertions.assertThat;
import static org.mockito.Mockito.mock;
import static org.mockito.Mockito.when;

/**
 * Unit test for the EXP-05 source_entity NOT IN denylist clause in RetrievalFilterBuilder.
 * Tests that: (1) non-empty denylist produces a nin clause on SOURCE_ENTITY,
 * (2) empty denylist produces no extra clause, (3) admin bypass returns null regardless.
 */
class RetrievalFilterBuilderDenylistTest {

    @Mock private AiAgentRagProperties ragProps;
    @Mock private AiAgentEmbeddingProperties embeddingProps;
    @Mock private LlmExposurePolicy llmExposurePolicy;

    private RetrievalFilterBuilder builder;

    @BeforeEach
    void setUp() {
        MockitoAnnotations.openMocks(this);
        when(ragProps.isAdminBypass()).thenReturn(true);
        when(embeddingProps.resolvedModel()).thenReturn("test-model");
        builder = new RetrievalFilterBuilder(ragProps, embeddingProps, llmExposurePolicy);
    }

    @Test
    void whenDenylistNonEmpty_thenFilterContainsNinClause() {
        when(llmExposurePolicy.getDenylistedEntityNames()).thenReturn(Set.of("OrderEntity"));

        Authentication auth = mock(Authentication.class);
        when(auth.getAuthorities()).thenReturn(List.of(
                new SimpleGrantedAuthority("ROLE_USER_ROLE"))); // non-admin

        Filter.Expression expr = builder.buildFor(auth);
        assertThat(expr).isNotNull();
        String exprStr = expr.toString();
        assertThat(exprStr)
                .as("Filter expression should contain source_entity nin clause")
                .contains(ChunkMetadata.SOURCE_ENTITY);
        // Assert "NIN" or "nin" or "NOT_IN" present in some form
        assertThat(exprStr.toUpperCase())
                .as("Filter expression should contain NIN operator")
                .containsAnyOf("NIN", "NOT IN", "NOT_IN");
    }

    @Test
    void whenDenylistEmpty_thenNoSourceEntityClause() {
        when(llmExposurePolicy.getDenylistedEntityNames()).thenReturn(Set.of());

        Authentication auth = mock(Authentication.class);
        when(auth.getAuthorities()).thenReturn(List.of(
                new SimpleGrantedAuthority("ROLE_USER_ROLE")));

        Filter.Expression expr = builder.buildFor(auth);
        // With empty denylist, expression should be the standard role-based filter only
        if (expr != null) {
            assertThat(expr.toString())
                    .as("Filter expression should not contain source_entity when denylist is empty")
                    .doesNotContain(ChunkMetadata.SOURCE_ENTITY);
        }
    }

    @Test
    void whenAdminUser_thenBuildForReturnsNull() {
        when(llmExposurePolicy.getDenylistedEntityNames()).thenReturn(Set.of("OrderEntity"));

        Authentication auth = mock(Authentication.class);
        when(auth.getAuthorities()).thenReturn(List.of(
                new SimpleGrantedAuthority("ROLE_" + AiAgentAdminRole.CODE.toUpperCase().replace('-', '_'))));

        Filter.Expression expr = builder.buildFor(auth);
        assertThat(expr).isNull(); // admin bypass — null means no filter
    }
}
```

Note: The exact authority string for admin bypass depends on how `toRoleCode()` converts it back — verify by reading RetrievalFilterBuilder.toRoleCode() and AiAgentAdminRole.CODE. The admin check looks for `roles.contains(AiAgentAdminRole.CODE)` after conversion. Adjust the test authority string accordingly.
