```java
/**
 * Persist permission/source-entity changes to the document row and schedule a reingest.
 * Atomic from the caller's perspective: save runs first; reingest enqueue runs second.
 * On reingest enqueue failure, the save is NOT rolled back (admin can retry via the row action).
 *
 * @return UpdatePermissionsResult with status (SAVED_AND_REINGESTING | SAVED_REINGEST_FAILED)
 */
public UpdatePermissionsResult updatePermissionsAndReingest(UUID documentId,
                                                            Collection<String> allowedRoles,
                                                            @Nullable String sourceEntityName) {
    AiKnowledgeDocument doc = dataManager.load(AiKnowledgeDocument.class)
        .id(documentId)
        .fetchPlan(FetchPlan.BASE)
        .one();
    // Set fields
    doc.setAllowedRolesJson(/* serialize allowedRoles */);
    if (sourceEntityName != null) {
        doc.setSourceEntityName(sourceEntityName);
    } else {
        doc.setSourceEntityName(null);
    }
    dataManager.save(doc);                      // FIRST: persist metadata
    try {
        reingest(documentId);                   // SECOND: schedule reingest
        return new UpdatePermissionsResult(UpdatePermissionsResult.Status.SAVED_AND_REINGESTING);
    } catch (Exception e) {
        // Save committed; reingest scheduling failed.
        // Do NOT roll back — document metadata is correct; user can retry via row action.
        return new UpdatePermissionsResult(UpdatePermissionsResult.Status.SAVED_REINGEST_FAILED);
    }
}
```
