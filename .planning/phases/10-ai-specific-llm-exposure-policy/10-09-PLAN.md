CRITICAL (Fix R7): Use `com.vaadin.flow.component.grid.Grid<Document>` — NOT `io.jmix.flowui.component.grid.DataGrid`. Declare it as a class field `private Grid<Document> chunksGrid;` and initialize it in `onInit`. Add columns programmatically with `addColumn` — no XML column declarations.

```java
package com.vn.agent.view.vectorstore;

import com.vaadin.flow.component.button.Button;
import com.vaadin.flow.component.button.ButtonVariant;
import com.vaadin.flow.component.dialog.Dialog;
import com.vaadin.flow.component.grid.Grid;
import com.vaadin.flow.component.html.Span;
import com.vaadin.flow.component.orderedlayout.HorizontalLayout;
import com.vaadin.flow.component.orderedlayout.VerticalLayout;
import com.vaadin.flow.component.textfield.TextArea;
import com.vaadin.flow.component.textfield.TextField;
import com.vaadin.flow.router.Route;
import io.jmix.core.Messages;
import io.jmix.flowui.component.textfield.TypedTextField;
import io.jmix.flowui.kit.component.button.JmixButton;
import io.jmix.flowui.view.*;
import org.springframework.ai.document.Document;
import org.springframework.ai.vectorstore.SearchRequest;
import org.springframework.ai.vectorstore.VectorStore;
import org.springframework.ai.vectorstore.filter.Filter;
import org.springframework.ai.vectorstore.filter.FilterExpressionTextParser;
import org.springframework.beans.factory.annotation.Autowired;
import com.vaadin.flow.component.orderedlayout.VerticalLayout;

import java.util.List;

@Route(value = "ai-agent/vector-store-debug", layout = DefaultMainViewParent.class)
@ViewController(id = "AiAgent_VectorStoreDebug")
@ViewDescriptor(path = "vector-store-debug-view.xml")
public class VectorStoreDebugView extends StandardView {

    private static final int DEFAULT_TOP_K = 100;

    @Autowired private VectorStore vectorStore;
    @Autowired private Messages messages;

    @ViewComponent private TextField metadataFilterField;
    @ViewComponent private Button filterClearBtn;
    @ViewComponent private Button filterHelpBtn;
    @ViewComponent private VerticalLayout chunksContainer;

    // Fix R7: plain Vaadin Grid<Document> — NOT Jmix DataGrid
    private Grid<Document> chunksGrid;

    @Subscribe
    public void onInit(final InitEvent event) {
        // Accessibility: aria labels for icon-only buttons (UI-SPEC Surface 4 requirement)
        filterClearBtn.setAriaLabel(messages.getMessage(getClass(), "vectorStoreDebug.filter.clear"));
        filterHelpBtn.setAriaLabel(messages.getMessage(getClass(), "vectorStoreDebug.filter.help"));

        // Fix R7: programmatic Grid<Document> — columns added programmatically, no metaclass binding
        chunksGrid = new Grid<>(Document.class, false);
        chunksGrid.setWidth("100%");
        chunksGrid.setMinHeight("20em");

        // Column 1: id
        chunksGrid.addColumn(Document::getId)
                .setKey("id")
                .setHeader(messages.getMessage(getClass(), "vectorStoreDebug.column.id"))
                .setWidth("260px")
                .setFlexGrow(0)
                .setResizable(true);

        // Column 2: content (truncated + expand button)
        chunksGrid.addComponentColumn(this::buildContentCell)
                .setKey("content")
                .setHeader(messages.getMessage(getClass(), "vectorStoreDebug.column.content"))
                .setResizable(true);

        // Column 3: metadata (JSON preview + expand button)
        chunksGrid.addComponentColumn(this::buildMetadataCell)
                .setKey("metadata")
                .setHeader(messages.getMessage(getClass(), "vectorStoreDebug.column.metadata"))
                .setResizable(true);

        chunksContainer.add(chunksGrid);
    }

    @Subscribe("searchBtn")
    public void onSearchClick(final ClickEvent<?> event) {
        metadataFilterField.setErrorMessage(null);
        String filterText = metadataFilterField.getValue();
        SearchRequest.Builder req = SearchRequest.builder()
                .query("")
                .topK(DEFAULT_TOP_K)
                .similarityThreshold(0.0);

        if (filterText != null && !filterText.isBlank()) {
            try {
                Filter.Expression expr = new FilterExpressionTextParser().parse(filterText);
                req.filterExpression(expr);
            } catch (Exception e) {
                metadataFilterField.setErrorMessage(
                        messages.getMessage(getClass(), "vectorStoreDebug.error.filterParse"));
                return;
            }
        }

        List<Document> docs = vectorStore.similaritySearch(req.build());
        chunksGrid.setItems(docs != null ? docs : List.of());
    }

    @Subscribe("filterClearBtn")
    public void onFilterClear(final ClickEvent<?> event) {
        metadataFilterField.clear();
        metadataFilterField.setErrorMessage(null);
    }

    private com.vaadin.flow.component.Component buildContentCell(Document doc) {
        String full = doc.getText() != null ? doc.getText() : "";
        String preview = full.length() > 120 ? full.substring(0, 120) + "..." : full;
        Span span = new Span(preview);
        if (full.length() > 120) {
            Button expand = new Button(messages.getMessage(getClass(), "vectorStoreDebug.action.expand"));
            expand.addThemeVariants(ButtonVariant.LUMO_TERTIARY_INLINE);
            expand.addClickListener(e -> openChunkDetail(doc));
            HorizontalLayout layout = new HorizontalLayout(span, expand);
            layout.setAlignItems(com.vaadin.flow.component.orderedlayout.FlexComponent.Alignment.CENTER);
            return layout;
        }
        return span;
    }

    private com.vaadin.flow.component.Component buildMetadataCell(Document doc) {
        String metaStr = doc.getMetadata() != null ? doc.getMetadata().toString() : "{}";
        String preview = metaStr.length() > 80 ? metaStr.substring(0, 80) + "..." : metaStr;
        Span span = new Span(preview);
        if (metaStr.length() > 80) {
            Button expand = new Button(messages.getMessage(getClass(), "vectorStoreDebug.action.expand"));
            expand.addThemeVariants(ButtonVariant.LUMO_TERTIARY_INLINE);
            expand.addClickListener(e -> openChunkDetail(doc));
            HorizontalLayout layout = new HorizontalLayout(span, expand);
            return layout;
        }
        return span;
    }

    private void openChunkDetail(Document doc) {
        // Standard Vaadin Dialog — not Jmix DialogWindows (which requires entity)
        Dialog dialog = new Dialog();
        dialog.setHeaderTitle(messages.getMessage(getClass(), "vectorStoreDebug.detail.title"));

        TextArea contentArea = new TextArea(messages.getMessage(getClass(), "vectorStoreDebug.column.content"));
        contentArea.setValue(doc.getText() != null ? doc.getText() : "");
        contentArea.setMinRows(5);
        contentArea.setMaxRows(20);
        contentArea.setReadOnly(true);
        contentArea.setWidth("100%");

        TextArea metaArea = new TextArea(messages.getMessage(getClass(), "vectorStoreDebug.column.metadata"));
        metaArea.setValue(doc.getMetadata() != null ? doc.getMetadata().toString() : "{}");
        metaArea.setMinRows(5);
        metaArea.setMaxRows(10);
        metaArea.setReadOnly(true);
        metaArea.setWidth("100%");

        VerticalLayout content = new VerticalLayout(contentArea, metaArea);
        content.setWidth("600px");
        dialog.add(content);

        Button closeBtn = new Button(
                messages.getMessage(getClass(), "vectorStoreDebug.detail.close"),
                e -> dialog.close());
        dialog.getFooter().add(closeBtn);
        dialog.open();
    }
}
```
