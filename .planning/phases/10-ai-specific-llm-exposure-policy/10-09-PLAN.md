```java
package com.vn.agent.view.vectorstore;

import com.vaadin.flow.component.button.Button;
import com.vaadin.flow.component.grid.Grid;
import com.vaadin.flow.component.textfield.TextField;
import com.vaadin.flow.data.renderer.ComponentRenderer;
import com.vaadin.flow.router.Route;
import io.jmix.core.Messages;
import io.jmix.flowui.component.grid.DataGrid;
import io.jmix.flowui.kit.action.ActionPerformedEvent;
import io.jmix.flowui.view.*;
import org.springframework.ai.document.Document;
import org.springframework.ai.vectorstore.SearchRequest;
import org.springframework.ai.vectorstore.VectorStore;
import org.springframework.ai.vectorstore.filter.Filter;
import org.springframework.ai.vectorstore.filter.FilterExpressionTextParser;
import org.springframework.beans.factory.annotation.Autowired;

import java.util.List;

@Route(value = "ai-agent/vector-store-debug", layout = DefaultMainViewParent.class)
@ViewController(id = "AiAgent_VectorStoreDebug")
@ViewDescriptor(path = "vector-store-debug-view.xml")
public class VectorStoreDebugView extends StandardView {

    private static final int DEFAULT_TOP_K = 100;

    @Autowired private VectorStore vectorStore;
    @Autowired private Messages messages;

    @ViewComponent private TextField metadataFilterField;
    @ViewComponent private DataGrid<Document> chunksDataGrid;
    @ViewComponent private Button filterClearBtn;
    @ViewComponent private Button filterHelpBtn;

    @Subscribe
    public void onInit(final InitEvent event) {
        // Accessibility: aria labels for icon-only buttons (UI-SPEC Surface 4 requirement)
        filterClearBtn.setAriaLabel(messages.getMessage(getClass(), "vectorStoreDebug.filter.clear"));
        filterHelpBtn.setAriaLabel(messages.getMessage(getClass(), "vectorStoreDebug.filter.help"));

        // Configure grid DataProvider — programmatic, no Jmix dataContainer
        chunksDataGrid.addColumn(Document::getId)
                .setKey("id").setHeader(messages.getMessage(getClass(), "vectorStoreDebug.column.id"))
                .setWidth("260px").setFlexGrow(0);
        chunksDataGrid.addComponentColumn(doc -> buildContentCell(doc))
                .setKey("content").setHeader(messages.getMessage(getClass(), "vectorStoreDebug.column.content"));
        chunksDataGrid.addComponentColumn(doc -> buildMetadataCell(doc))
                .setKey("metadata").setHeader(messages.getMessage(getClass(), "vectorStoreDebug.column.metadata"));
    }

    @Subscribe("searchBtn")
    public void onSearchClick(final ActionPerformedEvent event) {
        metadataFilterField.setErrorMessage(null);
        String filterText = metadataFilterField.getValue();
        SearchRequest.Builder req = SearchRequest.builder()
                .query("") // empty string — semantic order is meaningless; threshold=0.0 accepts all
                .topK(DEFAULT_TOP_K)
                .similarityThreshold(0.0);

        if (filterText != null && !filterText.isBlank()) {
            try {
                Filter.Expression expr = new FilterExpressionTextParser().parse(filterText);
                req.filterExpression(expr);
            } catch (Exception e) {
                metadataFilterField.setErrorMessage(
                        messages.getMessage(getClass(), "vectorStoreDebug.error.filterParse"));
                return;
            }
        }

        List<Document> docs = vectorStore.similaritySearch(req.build());
        chunksDataGrid.setItems(docs != null ? docs : List.of());
    }

    @Subscribe("filterClearBtn")
    public void onFilterClear(final ActionPerformedEvent event) {
        metadataFilterField.clear();
        metadataFilterField.setErrorMessage(null);
    }

    private com.vaadin.flow.component.Component buildContentCell(Document doc) {
        String full = doc.getText() != null ? doc.getText() : "";
        String preview = full.length() > 120 ? full.substring(0, 120) + "..." : full;
        com.vaadin.flow.component.html.Span span = new com.vaadin.flow.component.html.Span(preview);
        // Add expand button if content exceeds preview length
        if (full.length() > 120) {
            com.vaadin.flow.component.button.Button expand = new com.vaadin.flow.component.button.Button(
                    messages.getMessage(getClass(), "vectorStoreDebug.action.expand"));
            expand.addThemeVariants(com.vaadin.flow.component.button.ButtonVariant.LUMO_TERTIARY_INLINE);
            expand.addClickListener(e -> openChunkDetail(doc));
            com.vaadin.flow.component.orderedlayout.HorizontalLayout layout =
                    new com.vaadin.flow.component.orderedlayout.HorizontalLayout(span, expand);
            layout.setAlignItems(com.vaadin.flow.component.orderedlayout.FlexComponent.Alignment.CENTER);
            return layout;
        }
        return span;
    }

    private com.vaadin.flow.component.Component buildMetadataCell(Document doc) {
        String metaStr = doc.getMetadata() != null ? doc.getMetadata().toString() : "{}";
        String preview = metaStr.length() > 80 ? metaStr.substring(0, 80) + "..." : metaStr;
        com.vaadin.flow.component.html.Span span = new com.vaadin.flow.component.html.Span(preview);
        if (metaStr.length() > 80) {
            com.vaadin.flow.component.button.Button expand = new com.vaadin.flow.component.button.Button(
                    messages.getMessage(getClass(), "vectorStoreDebug.action.expand"));
            expand.addThemeVariants(com.vaadin.flow.component.button.ButtonVariant.LUMO_TERTIARY_INLINE);
            expand.addClickListener(e -> openChunkDetail(doc));
            com.vaadin.flow.component.orderedlayout.HorizontalLayout layout =
                    new com.vaadin.flow.component.orderedlayout.HorizontalLayout(span, expand);
            return layout;
        }
        return span;
    }

    private void openChunkDetail(Document doc) {
        // Open a simple dialog with full content and metadata text areas
        // Uses Vaadin Dialog directly (read-only detail, not Jmix DialogWindows which requires entity)
        com.vaadin.flow.component.dialog.Dialog dialog = new com.vaadin.flow.component.dialog.Dialog();
        dialog.setHeaderTitle(messages.getMessage(getClass(), "vectorStoreDebug.detail.title"));

        com.vaadin.flow.component.textfield.TextArea contentArea =
                new com.vaadin.flow.component.textfield.TextArea(
                        messages.getMessage(getClass(), "vectorStoreDebug.column.content"));
        contentArea.setValue(doc.getText() != null ? doc.getText() : "");
        contentArea.setMinRows(5); contentArea.setMaxRows(20);
        contentArea.setReadOnly(true); contentArea.setWidth("100%");

        com.vaadin.flow.component.textfield.TextArea metaArea =
                new com.vaadin.flow.component.textfield.TextArea(
                        messages.getMessage(getClass(), "vectorStoreDebug.column.metadata"));
        metaArea.setValue(doc.getMetadata() != null ? doc.getMetadata().toString() : "{}");
        metaArea.setMinRows(5); metaArea.setMaxRows(10);
        metaArea.setReadOnly(true); metaArea.setWidth("100%");

        com.vaadin.flow.component.orderedlayout.VerticalLayout content =
                new com.vaadin.flow.component.orderedlayout.VerticalLayout(contentArea, metaArea);
        content.setWidth("600px");
        dialog.add(content);

        com.vaadin.flow.component.button.Button closeBtn = new com.vaadin.flow.component.button.Button(
                messages.getMessage(getClass(), "vectorStoreDebug.detail.close"),
                e -> dialog.close());
        dialog.getFooter().add(closeBtn);
        dialog.open();
    }
}
```

Note on column setup in onInit vs XML: since the XML has `<column key="id">` etc. but no dataContainer, the XML columns serve as placeholders.
