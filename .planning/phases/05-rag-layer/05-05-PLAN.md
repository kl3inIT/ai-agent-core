            DockerImageName.parse("pgvector/pgvector:pg16").asCompatibleSubstituteFor("postgres"))
        .withDatabaseName("ai_agent_test")
        .withUsername("test").withPassword("test");

    static void props(DynamicPropertyRegistry r) {

        r.add("spring.datasource.username", PG::getUsername);
        r.add("spring.datasource.password", PG::getPassword);
        // Ensure pgvector extension + Jmix migrations run against this container
        r.add("spring.liquibase.enabled", () -> "true");
        // Use stub embeddings — no external API calls during tests
        r.add("jmix.ai-agent.embedding.model", () -> "stub-test-model");
    }

    @Autowired protected KnowledgeDocumentUploadService uploadService;
    @Autowired protected KnowledgeDocumentService documentService;
    @Autowired protected DataManager dataManager;
    @Autowired protected VectorStore vectorStore;
    @Autowired protected Metadata metadata;

    protected UUID uploadAndAwaitReady(String sourceUri, List<String> roles) {
        AiKnowledgeDocument saved = uploadService.upload(sourceUri, "CLASSPATH", roles);

        assertThat(reloaded.getStatus()).isEqualTo(IngestStatus.READY);
        return saved.getId();
    }
}

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| test → production code | Tests must not weaken production invariants by stubbing past security; stub embedding + sync executor are safe no-ops; a stubbed VectorStore used in rollback tests is explicit and @DirtiesContext-scoped |
| Testcontainers → host network | Container requires Docker; CI must provide it. Offline dev failure mode is acceptable — flagged in README or test `@EnabledIfDockerAvailable` |
