    ```java
    /**
     * Passthrough bean (D-02). The OpenAI starter auto-configures EmbeddingModel; this
     * @ConditionalOnMissingBean declaration is a no-op by default but gives hosts a
     * documented override seam — declare a host @Bean EmbeddingModel and this method
     * is skipped. Bean-collision is caught by EmbeddingModelBeanCollisionTest (RAG-02).
     *
     * <p>Intentionally accepts EmbeddingModel by parameter (not produced fresh) — this is
     * a passthrough, not a construction site.</p>
     */
    @Bean
    @ConditionalOnMissingBean
    public EmbeddingModel aiAgentEmbeddingModel(EmbeddingModel autoConfiguredEmbeddingModel) {
        return autoConfiguredEmbeddingModel;
    }

    /**
     * PgVectorStore bound to the Phase 2 Liquibase-owned table. initializeSchema=false
     * so Liquibase remains sole DDL owner (RESEARCH Pitfall #2). Dimension pinned to
     * match Phase 2 vector(1536) DDL (D-01).
     */
    @Bean
    @ConditionalOnMissingBean
    public VectorStore aiAgentVectorStore(JdbcTemplate jdbcTemplate,
                                          EmbeddingModel embeddingModel,
                                          AiAgentEmbeddingProperties embeddingProps) {
        return PgVectorStore.builder(jdbcTemplate, embeddingModel)
                .vectorTableName("AI_AGENT_KB_VECTOR_STORE")
                .initializeSchema(false)
                .dimensions(embeddingProps.resolvedDimensions())
                .build();
    }
    ```

    ```java
    import org.springframework.ai.embedding.EmbeddingModel;
    import org.springframework.ai.embedding.EmbeddingRequest;
    import org.springframework.ai.embedding.EmbeddingResponse;
    import org.springframework.ai.embedding.Embedding;

    import org.springframework.context.annotation.Bean;

    @TestConfiguration

        @Bean
        @Primary
        public EmbeddingModel stubEmbeddingModel() {
            return new EmbeddingModel() {
                @Override public EmbeddingResponse call(EmbeddingRequest request) {
                    List<Embedding> embeddings = request.getInstructions().stream()
                            .map(text -> new Embedding(deterministicVector(text), 0))
                            .toList();
                    return new EmbeddingResponse(embeddings);
                }
                @Override public float[] embed(Document document) { return deterministicVector(document.getText()); }
                private float[] deterministicVector(String text) {
                    float[] v = new float[1536];
                    int hash = text == null ? 0 : text.hashCode();
                    for (int i = 0; i < v.length; i++) v[i] = ((hash + i) % 1000) / 1000.0f;
                    return v;
                }
            };
    ```
