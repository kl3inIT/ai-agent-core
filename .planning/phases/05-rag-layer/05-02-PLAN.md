    package com.vn.agent.rag;

    import com.vn.agent.rag.config.AiAgentEmbeddingProperties;
    import com.vn.agent.rag.config.AiAgentRagProperties;
    import com.vn.agent.rag.ingest.ChunkMetadata;
    import com.vn.agent.security.AiAgentAdminRole;

    import org.springframework.security.core.Authentication;
    import org.springframework.security.core.GrantedAuthority;
    import org.springframework.stereotype.Component;

    import java.util.Locale;
    import java.util.Set;
    import java.util.stream.Collectors;

    /**
     * Pure-function bean: {@code Filter.Expression buildFor(Authentication)} — D-11.
     * <ul>
     *   <li>Admin with bypass ON → {@code null} (D-06). Caller MUST skip setting the
     *       {@code VectorStoreDocumentRetriever.FILTER_EXPRESSION} advisor param.</li>
     *   <li>Non-admin → {@code eq(embeddingModel, current) AND (OR over role_* flags)} (D-03 + D-09).</li>
     *   <li>Empty roles → a filter that matches zero chunks (fail-closed, D-05).</li>
     * </ul>
     */
    @Component
    public class RetrievalFilterBuilder {

        private final AiAgentRagProperties ragProps;
        private final AiAgentEmbeddingProperties embeddingProps;

        public RetrievalFilterBuilder(AiAgentRagProperties ragProps,
                                      AiAgentEmbeddingProperties embeddingProps) {
            this.ragProps = ragProps;
            this.embeddingProps = embeddingProps;
        }

        public Filter.Expression buildFor(Authentication auth) {
            Set<String> roles = auth == null ? Set.of() : auth.getAuthorities().stream()
                    .map(GrantedAuthority::getAuthority)
                    .collect(Collectors.toUnmodifiableSet());

            // D-06: admin bypass if flag enabled and caller holds the admin role code.
            if (ragProps.isAdminBypass() && roles.contains(AiAgentAdminRole.CODE)) {
                return null;
            }

            FilterExpressionBuilder b = new FilterExpressionBuilder();
            String currentModel = embeddingProps.resolvedModel();
            Filter.Expression modelPin = b.eq(ChunkMetadata.EMBEDDING_MODEL, currentModel).build();

            // D-05 fail-closed: empty role set → impossible match (no chunk stores documentId=__none__).
            if (roles.isEmpty()) {
                Filter.Expression unreachable = b.eq(ChunkMetadata.DOCUMENT_ID, "__none__").build();
                return b.and(modelPin, unreachable).build();
            }

            // D-09 ANY semantics via Option A flattened role flags (RESEARCH Pitfall #1 mitigation).
            Filter.Expression roleOverlap = null;
            for (String role : roles) {
                String key = ChunkMetadata.ROLE_FLAG_PREFIX + role.toLowerCase(Locale.ROOT);
                Filter.Expression clause = b.eq(key, true).build();
                roleOverlap = (roleOverlap == null) ? clause : b.or(roleOverlap, clause).build();
            }

            return b.and(modelPin, roleOverlap).build();
        }
    }
    ```

    package com.vn.agent.rag.advisor;

    import com.vn.agent.rag.config.AiAgentRagProperties;
    import org.springframework.ai.rag.advisor.RetrievalAugmentationAdvisor;
    import org.springframework.ai.rag.retrieval.search.VectorStoreDocumentRetriever;
    import org.springframework.ai.vectorstore.VectorStore;
    import org.springframework.boot.autoconfigure.condition.ConditionalOnMissingBean;
    import org.springframework.context.annotation.Bean;
    import org.springframework.context.annotation.Configuration;
    import org.springframework.core.Ordered;

    /**
     * Produces the @Bean RetrievalAugmentationAdvisor slotted between MessageChatMemoryAdvisor (+200)
     * and ToolCallAdvisor (+300) per Phase 4 D-02. Retriever is built with NO static filter —
     * the per-request FILTER_EXPRESSION (set by DefaultChatServiceImpl from RetrievalFilterBuilder)
     * is the sole security-critical filter (RESEARCH Pitfall #3: a static retriever filter is
     * REPLACED, not AND-ed, by the runtime param).
     */
    @Configuration
    public class RetrievalAugmentationAdvisorFactory {

        public static final int ADVISOR_ORDER = Ordered.HIGHEST_PRECEDENCE + 250;

        @Bean
        @ConditionalOnMissingBean
        public RetrievalAugmentationAdvisor retrievalAugmentationAdvisor(
                VectorStore vectorStore, AiAgentRagProperties props) {
            VectorStoreDocumentRetriever retriever = VectorStoreDocumentRetriever.builder()
                    .vectorStore(vectorStore)
                    .similarityThreshold(props.resolvedSimilarityThreshold())
                    .topK(props.resolvedTopK())
                    .build();   // NO .filterExpression(...) — Pitfall #3
            return RetrievalAugmentationAdvisor.builder()
                    .documentRetriever(retriever)
                    .order(ADVISOR_ORDER)
                    .build();
        }
    }
    ```
